function handleRequest(e) {

    if (e.postData && e.postData.contents) {

    // não deve ir pro código público dele. A ação só age sobre um contato que

    // Firebase de quem está logado; a checagem fica dentro de cada ação.

    var action = params.action || body.action || '';

    if (action === 'setOverride')     return jsonOutput(setOverride(body));

// ── Helpers de aba ──────────────────────────────────────────────────

function getSheet(name) {
  var ss    = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(name) || findSheetLoose(ss, name);
  if (!sheet) throw new Error('Aba "' + name + '" não encontrada na planilha.');
  return sheet;
}

function getSheetSafe(name) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  return ss.getSheetByName(name) || findSheetLoose(ss, name);
}

function findSheetLoose(ss, name) {
  var target = normalizeSheetName(name);
  var sheets = ss.getSheets();
  for (var i = 0; i < sheets.length; i++) {
    if (normalizeSheetName(sheets[i].getName()) === target) return sheets[i];
  }
  return null;
}

function normalizeSheetName(s) {
  return String(s || '')
    .trim().toLowerCase()
    .normalize('NFD').replace(/[\u0300-\u036f]/g, '')
    .replace(/[_\s]+/g, ' ').trim();
}

// ── Leitura de planilha ─────────────────────────────────────────────

function sheetToObjects(sheet) {
  var values = sheet.getDataRange().getValues();
  if (values.length === 0) return [];
  var headers = values[0].map(function (h) { return String(h).trim(); });
  var out = [];
  for (var i = 1; i < values.length; i++) {
    var row = values[i];
    if (row.every(function (c) { return c === '' || c === null; })) continue;
    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      if (!headers[j]) continue;
      var val = row[j];
      if (isDateLike(val)) val = formatDateBR(val);
      obj[headers[j]] = val;
    }
    out.push(obj);
  }
  return out;
}

/**
 * Equivalente ao "cache em memória tipo Redis" nesse ecossistema: o Apps
 * Script tem um CacheService nativo, compartilhado entre TODAS as
 * execuções do script (não só a atual), com expiração automática — exatamente
 * o papel que um Redis cumpriria aqui. Usado só pra abas que são lidas o
 * tempo todo e mudam pouco (Vendedores, Serviços) — não faz sentido pra
 * abas que mudam a cada request (Agendamentos, Vendas, Funil).
 *
 * Sempre que uma dessas abas for escrita, chame invalidarCache(chave) logo
 * depois — senão quem ler nos próximos minutos vai ver dado desatualizado.
 */
var SG_CACHE_TTL_PADRAO = 300; // 5 minutos — equilíbrio entre "atualizado" e "rápido"

function getSheetObjectsComCache(sheetName, chave, ttlSegundos) {
  var cache = CacheService.getScriptCache();
  var cacheado = cache.get(chave);
  if (cacheado) {
    try { return JSON.parse(cacheado); } catch (e) { /* cache corrompido — ignora e recarrega da planilha */ }
  }
  var sheet = getSheetSafe(sheetName);
  var dados = sheet ? sheetToObjects(sheet) : [];
  if (normalizeSheetName(sheetName) === normalizeSheetName(CONFIG.ABA_VENDEDORES)) {
    dados = sanitizeVendedoresPublico(dados);
  }
  try {
    var json = JSON.stringify(dados);
    if (json.length < 95000) cache.put(chave, json, ttlSegundos || SG_CACHE_TTL_PADRAO); // CacheService tem limite de ~100KB por chave
  } catch (e) { /* não é crítico — só não guarda cache dessa vez, continua funcionando normal */ }
  return dados;
}

/**
 * Remove os campos sensíveis de autenticação (hash/salt de senha, token de
 * redefinição) antes de qualquer vendedor sair pro cliente. Toda leitura de
 * Vendedores por getSheetObjectsComCache passa por aqui — login/troca de
 * senha não usam esse caminho, então continuam funcionando normalmente.
 */
function sanitizeVendedoresPublico(lista) {
  var camposOcultos = { SenhaHash: 1, SenhaSalt: 1, ResetToken: 1, ResetTokenExpira: 1 };
  return (lista || []).map(function (v) {
    var out = {};
    Object.keys(v).forEach(function (k) { if (!camposOcultos[k]) out[k] = v[k]; });
    return out;
  });
}

function invalidarCache(chave) {
  try { CacheService.getScriptCache().remove(chave); } catch (e) { /* não é crítico */ }
}

function abonosSheetToObjects(sheet) {
  var values     = sheet.getDataRange().getValues();
  if (values.length === 0) return [];
  var headers    = values[0].map(function (h) { return String(h).trim(); });
  var horaFields = { Entrada: true, Almoco: true, RetornoAlmoco: true, Saida: true };
  var out = [];
  for (var i = 1; i < values.length; i++) {
    var row = values[i];
    if (row.every(function (c) { return c === '' || c === null; })) continue;
    var obj = {};
    for (var j = 0; j < headers.length; j++) {
      if (!headers[j]) continue;
      var val = row[j];
      if      (horaFields[headers[j]]) val = formatHoraSomente(val);
      else if (isDateLike(val))         val = formatDateBR(val);
      obj[headers[j]] = val;
    }
    out.push(obj);
  }
  return out;
}

function isDateLike(val) {
  return val instanceof Date ||
    (val && typeof val.getFullYear === 'function' && typeof val.getMonth === 'function');
}

function formatDateBR(d) {
  return String(d.getDate()).padStart(2,'0') + '/' +
    String(d.getMonth()+1).padStart(2,'0') + '/' + d.getFullYear() + ' ' +
    String(d.getHours()).padStart(2,'0') + ':' +
    String(d.getMinutes()).padStart(2,'0') + ':' +
    String(d.getSeconds()).padStart(2,'0');
}

function formatHoraSomente(val) {
  if (val === '' || val === null || val === undefined) return '';
  if (isDateLike(val)) {
    return String(val.getHours()).padStart(2,'0') + ':' + String(val.getMinutes()).padStart(2,'0');
  }
  var m = String(val).trim().match(/^(\d{1,2}):(\d{2})/);
  return m ? m[1].padStart(2,'0') + ':' + m[2] : '';
}

function normalizeDateKey(val) {
  if (isDateLike(val)) {
    return val.getFullYear() + '-' +
      String(val.getMonth()+1).padStart(2,'0') + '-' + String(val.getDate()).padStart(2,'0');
  }
  var str = String(val).trim();
  var iso = str.match(/^(\d{4})-(\d{2})-(\d{2})/);
  if (iso) return iso[1] + '-' + iso[2] + '-' + iso[3];
  var br  = str.match(/^(\d{1,2})\/(\d{1,2})\/(\d{4})/);
  if (br)  return br[3] + '-' + br[2].padStart(2,'0') + '-' + br[1].padStart(2,'0');
  return null;
}

var ABONOS_HEADERS = ['Funcionario','Data','Abonado','Entrada','Almoco','RetornoAlmoco','Saida'];

function getOrCreateAbonosSheet() {
  var ss    = SpreadsheetApp.getActiveSpreadsheet();
  var sheet = ss.getSheetByName(CONFIG.ABA_ABONOS);
  if (!sheet) {
    sheet = ss.insertSheet(CONFIG.ABA_ABONOS);
    sheet.getRange(1,1,1,7).setValues([ABONOS_HEADERS]);
    sheet.setFrozenRows(1);
  } else {
    var first = String(sheet.getRange(1,1).getValue()||'').trim();
    if (first !== 'Funcionario') {
      sheet.insertRowBefore(1);
      sheet.getRange(1,1,1,7).setValues([ABONOS_HEADERS]);
      sheet.setFrozenRows(1);
    }
  }
  sheet.getRange('B2:B').setNumberFormat('@STRING@');
  sheet.getRange('D2:G').setNumberFormat('@STRING@');
  return sheet;
}

/**
 * Gera um ID curto tipo "a3f9c21b" (8 caracteres hex)
 */
/**
 * Garante que um valor monetário/numérico vindo do front-end (que sempre
 * manda no formato americano — ponto decimal, ex: "39.9" — porque vem de
 * <input type="number">) seja gravado na planilha como NÚMERO de verdade,
 * nunca como texto. Se ficar como texto, o front-end (que trata qualquer
 * texto como formato brasileiro, ponto = milhar) lê errado depois —
 * foi exatamente isso que causou o bug de "39,90" virar "3990".
 */
/**
 * Pega só os dígitos do telefone (ignora hífen, espaço, parênteses, +55…) e
 * devolve os últimos 8 — usado pra detectar cliente duplicado mesmo quando
 * o número foi digitado em formatos diferentes. Espelha exatamente a mesma
 * lógica do front-end (window.SGUtil.ultimos8DigitosTelefone).
 */
function paraUltimos8Digitos(tel) {
  var digitos = String(tel || '').replace(/\D/g, '');
  return digitos.length >= 8 ? digitos.slice(-8) : null;
}

function paraNumero(v) {
  if (v === undefined || v === null || v === '') return '';
  if (typeof v === 'number') return v;
  var n = parseFloat(String(v).replace(',', '.'));
  return isNaN(n) ? '' : n;
}

function generateShortId() {
  return Utilities.getUuid().replace(/-/g,'').substring(0,8);
}

// ── Actions: Ponto ─────────────────────────────────────────────────

function getData() {
  var vendedores  = getSheetObjectsComCache(CONFIG.ABA_VENDEDORES, 'sh_vendedores');
  var ponto       = sheetToObjects(getSheet(CONFIG.ABA_PONTO));
  var abonosSheet = getOrCreateAbonosSheet();
  var abonosRaw   = abonosSheetToObjects(abonosSheet);

  var overrides = {};
  abonosRaw.forEach(function (row) {
    if (!row.Funcionario || !row.Data) return;
    var dataKey = normalizeDateKey(row.Data);
    if (!dataKey) return;
    overrides[String(row.Funcionario) + '|' + dataKey] = {
      abonado: String(row.Abonado).toUpperCase() === 'TRUE' || row.Abonado === true,
      entrada: row.Entrada || '', almoco: row.Almoco  || '',
      retorno: row.RetornoAlmoco || '', saida: row.Saida || ''
    };
  });

  return { ok: true, vendedores: vendedores, ponto: ponto, overrides: overrides,
           atualizadoEm: new Date().toISOString() };
}

function setOverride(body) {
  var funcionario = body.funcionario, data = body.data;
  if (!funcionario || !data) return { ok: false, erro: 'funcionario e data são obrigatórios.' };

  var sheet  = getOrCreateAbonosSheet();
  var values = sheet.getDataRange().getValues();
  var rowIdx = -1;
  for (var i = 1; i < values.length; i++) {
    if (String(values[i][0]) === String(funcionario) &&
        normalizeDateKey(values[i][1]) === String(data)) {
      rowIdx = i + 1; break;
    }
  }

  var abonado = !!body.abonado, entrada = body.entrada||'', almoco = body.almoco||'',
      retorno = body.retorno||'', saida = body.saida||'';

  if (!abonado && !entrada && !almoco && !retorno && !saida) {
    if (rowIdx !== -1) sheet.deleteRow(rowIdx);
    return { ok: true, acao: 'removido' };
  }

  var rowValues = [funcionario, data, abonado, entrada, almoco, retorno, saida];
  if (rowIdx !== -1) {

/**
 * action: salvarVenda — body { idVenda?, idCliente, idServico, idVendedor, dataVenda ('YYYY-MM-DD'), valor }
 * Cria ou atualiza uma linha na aba Vendas.
 */
function actionSalvarVenda(body) {
  var idCliente = body.idCliente, idServico = body.idServico, idVendedor = body.idVendedor,
      dataVenda = body.dataVenda, valor = body.valor;
  if (!idCliente || !idServico || !idVendedor || !dataVenda || valor === undefined || valor === '') {
    return { ok: false, erro: 'Cliente, serviço, vendedor, data e valor são obrigatórios.' };
  }
  var sheet = getSheetSafe(CONFIG.ABA_VENDAS);
  if (!sheet) return { ok: false, erro: 'Aba Vendas não encontrada.' };
  var data = body.idVenda ? sheet.getDataRange().getValues() : [getHeaderRow(sheet)];
  var headers = data.length ? data[0].map(function (h) { return String(h).trim(); }) : ['IdVenda', 'IdCliente', 'IdServico', 'IdVendedor', 'DataVenda', 'Valor'];
  var colIdx = {}; headers.forEach(function (h, i) { colIdx[h] = i + 1; });

  var rowIndex = -1;
  if (body.idVenda && colIdx['IdVenda']) {
    for (var i = 1; i < data.length; i++) {
      if (String(data[i][colIdx['IdVenda'] - 1]) === String(body.idVenda)) { rowIndex = i + 1; break; }
    }
  }

  var campos = {
    'IdCliente': idCliente, 'IdServico': idServico, 'IdVendedor': idVendedor,
    'DataVenda': parseDataISO(dataVenda), 'Valor': paraNumero(valor)
  };

  var novoId = body.idVenda;
  if (rowIndex === -1) {
    novoId = novoId || generateShortId(); // usa o ID que o cliente já gerou (otimista) — só gera aqui se não veio
    var lastCol = sheet.getLastColumn() || headers.length;
    var novaLinha = new Array(lastCol).fill('');
    Object.keys(colIdx).forEach(function (h) {
      if (h === 'IdVenda') novaLinha[colIdx[h] - 1] = novoId;
      else if (campos[h] !== undefined) novaLinha[colIdx[h] - 1] = campos[h];
    });
    sheet.getRange(sheet.getLastRow() + 1, 1, 1, novaLinha.length).setValues([novaLinha]);
  } else {
    Object.keys(campos).forEach(function (h) { if (colIdx[h]) sheet.getRange(rowIndex, colIdx[h]).setValue(campos[h]); });
  }
  return { ok: true, idVenda: novoId };
}

 * action: excluirVenda — body { idVenda }

      }

  return { ok: false, erro: 'Venda não encontrada.' };

/**
 * Sugestão de categoria da DRE pelo Jev (TypeSafe), 2026-10-13.
 * O front monta o pedido (js/dre-categorias.js, montarPedidoJev) e este código
 * só repassa, com a chave da TypeSafe nas Propriedades do script
 * (TYPESAFE_API_KEY, nunca no código nem no front: crença 13). Não dá para
 * checar administrador aqui (o front não manda solicitanteId ao Apps Script,
 * ver sg-auth.js), então a única barreira é a CHAVE_SECRETA do dispatcher, que
 * está no JS público. Por isso os limites para não virar proxy aberto de LLM:
 * modelo fixo, uma pergunta só, texto e pedido com tamanho máximo.
 * A descrição do custo (que pode ter nome de funcionário) vai para a TypeSafe.
 */
function actionSugerirCategoriaJev(body) {
  var chaveJev = PropertiesService.getScriptProperties().getProperty('TYPESAFE_API_KEY');
  if (!chaveJev) return { ok: false, erro: 'Jev não configurado.' };
  var pedido = body.pedido;
  if (!pedido || typeof pedido.state !== 'object' || !pedido.questions) {
    return { ok: false, erro: 'Pedido inválido.' };
  }
  var nomes = Object.keys(pedido.questions);
  if (nomes.length !== 1 || nomes[0] !== 'categoria' || pedido.questions.categoria.type !== 'choice') {
    return { ok: false, erro: 'Pedido inválido.' };
  }
  var payload = { model: 'jev-latest', state: pedido.state, questions: pedido.questions };
  var texto = JSON.stringify(payload);
  if (texto.length > 6000 || String(pedido.state.descricao || '').length > 300) {
    return { ok: false, erro: 'Pedido grande demais.' };
  }
  var resp = UrlFetchApp.fetch('https://api.typesafe.ai/v1/systemone', {
    method: 'post',
    contentType: 'application/json',
    headers: { Authorization: 'Bearer ' + chaveJev },
    payload: texto,
    muteHttpExceptions: true
  });
  var codigo = resp.getResponseCode();
  if (codigo !== 200) return { ok: false, erro: 'Jev respondeu ' + codigo + '.' };
  try {
    return { ok: true, resposta: JSON.parse(resp.getContentText()) };
  } catch (err) {
    return { ok: false, erro: 'Resposta do Jev ilegível.' };
  }
}

/**
 * action: salvarFunil — body { idOportunidade?, idCliente, idVendedor, idServico?,
 *   etapa, observacoes?, valorEstimado?, motivoPerda? }
 * Cria ou atualiza um lead do Funil. Se a etapa mudou (ou é lead novo), registra
 * a transição na aba FunilLog — é isso que alimenta o cálculo de "dias na
 * etapa" e a cor de SLA na tela, então precisa ficar consistente com a UI.
 */
function actionSalvarFunil(body) {
  var idCliente = body.idCliente, idVendedor = body.idVendedor, etapa = body.etapa;
  if (!idCliente || !idVendedor || !etapa) {
    return { ok: false, erro: 'Cliente, vendedor e etapa são obrigatórios.' };
  }

  var sheet = getSheetSafe(CONFIG.ABA_FUNIL);
  if (!sheet) return { ok: false, erro: 'Aba Funil não encontrada.' };
  var data = body.idOportunidade ? sheet.getDataRange().getValues() : [getHeaderRow(sheet)];
  var headers = data[0].map(function (h) { return String(h).trim(); });
  var colIdx = {}; headers.forEach(function (h, i) { colIdx[h] = i + 1; });

  var idCol = colIdx['IdOportunidade'];
  if (!idCol) return { ok: false, erro: 'Coluna IdOportunidade não encontrada na aba Funil.' };

  var rowIndex = -1, etapaAnterior = null;
  if (body.idOportunidade) {
    for (var i = 1; i < data.length; i++) {
      if (String(data[i][idCol - 1]) === String(body.idOportunidade)) {
        rowIndex = i + 1;
        etapaAnterior = colIdx['Etapa'] ? String(data[i][colIdx['Etapa'] - 1] || '').trim() : null;
        break;
      }
    }
  }

  var agora = new Date();
  var campos = {
    'IdCliente': idCliente,
    'IdVendedor': idVendedor,
    'IdServico': body.idServico || '',
    'Etapa': etapa,
    'Observacoes': body.observacoes || '',
    'Valor Estimado': body.valorEstimado !== undefined ? body.valorEstimado : '',
    'Motivo Perda': body.motivoPerda || ''
  };

  var novoId = body.idOportunidade;
  if (rowIndex === -1) {
    novoId = novoId || generateShortId(); // usa o ID que o cliente já gerou (otimista) — só gera aqui se não veio
    var lastCol = sheet.getLastColumn();
    var novaLinha = new Array(lastCol).fill('');
    Object.keys(colIdx).forEach(function (h) {
      if (h === 'IdOportunidade') novaLinha[colIdx[h] - 1] = novoId;
      else if (h === 'Data Criacao') novaLinha[colIdx[h] - 1] = agora;
      else if (campos[h] !== undefined) novaLinha[colIdx[h] - 1] = campos[h];
    });
    sheet.getRange(sheet.getLastRow() + 1, 1, 1, novaLinha.length).setValues([novaLinha]);
  } else {
    Object.keys(campos).forEach(function (h) {
      if (colIdx[h]) sheet.getRange(rowIndex, colIdx[h]).setValue(campos[h]);
    });
  }

  if (etapaAnterior === null || etapaAnterior !== etapa) {
    registrarFunilLog(novoId, etapaAnterior || '', etapa, agora);
  }

  return { ok: true, idOportunidade: novoId };
}

/**
 * Lê só a LINHA DE CABEÇALHO de uma aba, sem trazer os dados inteiros —
 * muito mais rápido que getDataRange() em abas grandes (Clientes, Ponto,
 * FunilLog, etc.), usado sempre que só precisamos saber em que coluna cada
 * campo está, não os dados já existentes.
 */
function getHeaderRow(sheet) {
  var lastCol = sheet.getLastColumn();
  if (lastCol === 0) return [];
  return sheet.getRange(1, 1, 1, lastCol).getValues()[0].map(function (h) { return String(h).trim(); });
}

function registrarFunilLog(idOportunidade, etapaAnterior, etapaNova, quando) {
  var logSheet = getSheetSafe(CONFIG.ABA_FUNIL_LOG);
  if (!logSheet) return;
  var headers = getHeaderRow(logSheet);
  var linha = headers.map(function (h) {
    if (h === 'IdOportunidade') return idOportunidade;
    if (h === 'EtapaAnterior') return etapaAnterior;
    if (h === 'EtapaNova') return etapaNova;
    if (h === 'Datahora' || h === 'DataHora') return quando;
    return '';
  });
  logSheet.getRange(logSheet.getLastRow() + 1, 1, 1, linha.length).setValues([linha]);
}

/**
 * action: excluirFunil — body { idOportunidade }
 * Remove a linha do lead da aba Funil. Não apaga o histórico em FunilLog
 * (fica como registro, não afeta nada já que o lead não existe mais na lista).
 */
function actionExcluirFunil(body) {
  var idOportunidade = body.idOportunidade;
  if (!idOportunidade) return { ok: false, erro: 'idOportunidade é obrigatório.' };
  var sheet = getSheetSafe(CONFIG.ABA_FUNIL);
  if (!sheet) return { ok: false, erro: 'Aba Funil não encontrada.' };
  var data = sheet.getDataRange().getValues();
  var idCol = -1, vendedorCol = -1;
  for (var c = 0; c < data[0].length; c++) {
    var h = String(data[0][c]).trim();
    if (h === 'IdOportunidade') idCol = c;
    if (h === 'IdVendedor') vendedorCol = c;
  }
  if (idCol === -1) return { ok: false, erro: 'Coluna IdOportunidade não encontrada.' };
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][idCol]) === String(idOportunidade)) {
      var dono = vendedorCol !== -1 ? String(data[i][vendedorCol]) : null;
      var ehDono = dono && String(body.solicitanteId) === dono;
      if (!ehDono && !isSolicitanteAdmin(body.solicitanteId)) {
        return { ok: false, erro: 'Você só pode excluir os próprios leads.' };
      }
      sheet.deleteRow(i + 1);
      return { ok: true };
    }
  }
  return { ok: false, erro: 'Lead não encontrado.' };
}

/**

  var rowIndex = -1;
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][colIdx['IdVendedor'] - 1]) === String(idVendedor) &&
        Number(data[i][colIdx['Ano'] - 1]) === Number(ano) &&
        Number(data[i][colIdx['Mes'] - 1]) === Number(mes)) { rowIndex = i + 1; break; }
  }

  var campos = { 'IdVendedor': idVendedor, 'Ano': ano, 'Mes': mes, 'ValorMeta': paraNumero(valorMeta) };
  var novoId;
  if (rowIndex === -1) {
    novoId = novoId || generateShortId(); // usa o ID que o cliente já gerou (otimista) — só gera aqui se não veio
    var lastCol = sheet.getLastColumn();

      if (h === 'IdMetaIndividual') novaLinha[colIdx[h] - 1] = novoId;

    });
    sheet.getRange(sheet.getLastRow() + 1, 1, 1, novaLinha.length).setValues([novaLinha]);
  } else {
    novoId = data[rowIndex - 1][colIdx['IdMetaIndividual'] - 1];
    Object.keys(campos).forEach(function (h) { if (colIdx[h]) sheet.getRange(rowIndex, colIdx[h]).setValue(campos[h]); });
  }
  return { ok: true, idMetaIndividual: novoId };

/**

 * Remove a sobreposição — o vendedor volta a usar a meta padrão (dividida
 * igualmente entre os ativos) pra esse mês.
 */
function actionExcluirMetaIndividual(body) {
  var idVendedor = body.idVendedor, ano = body.ano, mes = body.mes;
  if (!idVendedor || !ano || !mes) return { ok: false, erro: 'Vendedor, ano e mês são obrigatórios.' };
  var sheet = getSheetSafe(CONFIG.ABA_METAS_INDIVIDUAIS);
  if (!sheet) return { ok: true }; // nada pra remover
  var data = sheet.getDataRange().getValues();
  var headers = data[0].map(function (h) { return String(h).trim(); });
  var colIdx = {}; headers.forEach(function (h, i) { colIdx[h] = i + 1; });
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][colIdx['IdVendedor'] - 1]) === String(idVendedor) &&
        Number(data[i][colIdx['Ano'] - 1]) === Number(ano) &&
        Number(data[i][colIdx['Mes'] - 1]) === Number(mes)) {
      sheet.deleteRow(i + 1);
      return { ok: true };
    }
  }
  return { ok: true }; // já não existia, tudo bem
}

/**
 * action: salvarCliente — body { idCliente?, nome, tipoPessoa, cpfCnpj?, telefone,
 *   email?, endereco?, statusCliente?, vendedorResponsavel? }
 * Cria (sem idCliente EXISTENTE) ou atualiza (com idCliente já existente na
 * planilha) um cliente com TODOS os campos da aba Clientes. Usado tanto pela
 * tela CRUD de Clientes quanto pelo cadastro rápido dentro do Funil.
 *
 * IMPORTANTE: o navegador SEMPRE manda um idCliente (gerado na hora, pra
 * deixar a tela otimista/instantânea) — inclusive quando é um cliente NOVO.
 * Por isso a decisão create-vs-update aqui é baseada em ACHAR ou NÃO a linha
 * com esse ID na planilha, nunca em "o campo veio preenchido ou não".
 */
function actionSalvarCliente(body) {
  var nome = String(body.nome || '').trim();
  var telefone = String(body.telefone || '').trim();
  var tipoPessoa = String(body.tipoPessoa || '').trim();
  if (!nome || !telefone) return { ok: false, erro: 'Nome e telefone são obrigatórios.' };
  if (!tipoPessoa) return { ok: false, erro: 'Tipo de pessoa (Física/Jurídica) é obrigatório.' };

  var sheet = getSheetSafe(CONFIG.ABA_CLIENTES);
  if (!sheet) return { ok: false, erro: 'Aba Clientes não encontrada.' };
  var headers = getHeaderRow(sheet);
  var colIdx = {}; headers.forEach(function (h, i) { colIdx[h] = i + 1; });
  if (!colIdx['IdCliente']) return { ok: false, erro: 'Coluna IdCliente não encontrada.' };

  var campos = {};
  var nomeCol = colIdx['Nome Razao Social'] || colIdx['Nome'];
  if (nomeCol) campos[nomeCol] = nome;
  if (colIdx['Tipo Pessoa']) campos[colIdx['Tipo Pessoa']] = tipoPessoa;
  if (colIdx['CPF ou CNPJ'] && body.cpfCnpj !== undefined) campos[colIdx['CPF ou CNPJ']] = body.cpfCnpj;
  if (colIdx['Telefone']) campos[colIdx['Telefone']] = telefone;
  if (colIdx['Email'] && body.email !== undefined) campos[colIdx['Email']] = body.email;
  if (colIdx['Endereco'] && body.endereco !== undefined) campos[colIdx['Endereco']] = body.endereco;
  if (colIdx['Status Cliente'] && body.statusCliente) campos[colIdx['Status Cliente']] = body.statusCliente;
  if (colIdx['Vendedor Responsavel'] && body.vendedorResponsavel !== undefined) campos[colIdx['Vendedor Responsavel']] = body.vendedorResponsavel;
  if (colIdx['CPFEquatorial'] && body.cpfEquatorial !== undefined) campos[colIdx['CPFEquatorial']] = body.cpfEquatorial;
  if (colIdx['DataNascimentoEquatorial'] && body.dataNascimentoEquatorial) campos[colIdx['DataNascimentoEquatorial']] = parseDataISO(body.dataNascimentoEquatorial);

  // Procura a linha pelo ID recebido — SEM decidir create/update ainda.
  var rowIndex = -1;
  var data = sheet.getDataRange().getValues(); // sempre lê tudo — precisa pra achar a linha E pra checar telefone duplicado
  if (body.idCliente) {
    for (var i = 1; i < data.length; i++) {
      if (String(data[i][colIdx['IdCliente'] - 1]) === String(body.idCliente)) { rowIndex = i + 1; break; }
    }
  }

  // Só checa duplicidade em CRIAÇÃO de cliente novo (rowIndex===-1), e só
  // bloqueia se o endereço vier em branco — se a pessoa preencheu um
  // endereço, presume-se que ela já conferiu/sabe que é um contato
  // diferente. O front-end já faz essa mesma checagem antes de chamar essa
  // ação (pra dar feedback na hora); isso aqui é só uma segunda trava, caso
  // dois vendedores cadastrem quase ao mesmo tempo.
  if (rowIndex === -1 && !body.confirmarClienteDiferente) {
    var enderecoNovoVazio = !String(body.endereco || '').trim();
    if (enderecoNovoVazio && colIdx['Telefone']) {
      var alvo8 = paraUltimos8Digitos(telefone);
      if (alvo8) {
        for (var j = 1; j < data.length; j++) {
          if (paraUltimos8Digitos(data[j][colIdx['Telefone'] - 1]) === alvo8) {
            return { ok: false, erro: 'Já existe um contato cadastrado para esse número', duplicidadeTelefone: true };
          }
        }
      }
    }
  }

  if (rowIndex !== -1) {
    // Achou a linha — é edição de um cliente que já existe.
    Object.keys(campos).forEach(function (col) { sheet.getRange(rowIndex, Number(col)).setValue(campos[col]); });
    return { ok: true, idCliente: body.idCliente };
  }

  // Não achou (ou não veio ID nenhum) — cria uma linha nova, usando o ID que
  // o navegador já gerou (otimista); só gera um novo aqui se por algum
  // motivo não tiver vindo nenhum.
  var novoId = body.idCliente || generateShortId();
  var lastCol = sheet.getLastColumn();
  var novaLinha = new Array(lastCol).fill('');
  Object.keys(campos).forEach(function (col) { novaLinha[Number(col) - 1] = campos[col]; });
  novaLinha[colIdx['IdCliente'] - 1] = novoId;
  sheet.getRange(sheet.getLastRow() + 1, 1, 1, novaLinha.length).setValues([novaLinha]);
  return { ok: true, idCliente: novoId, nome: nome, telefone: telefone };
}

/**
 * Atualiza campos cadastrais de um cliente existente na aba Clientes,
 * localizando a linha pelo IdCliente (coluna A).
 * Só sobrescreve os campos que vierem definidos no body (nome/cpf/endereco),
 * preservando os demais valores da linha (Telefone, Email, Status, etc).
 */
function updateCliente(body) {
  var idCliente = body.idCliente;
  if (!idCliente) return { ok: false, erro: 'idCliente é obrigatório.' };

  var sheet = getSheetSafe(CONFIG.ABA_CLIENTES);
  if (!sheet) return { ok: false, erro: 'Aba Clientes não encontrada.' };

  var data = sheet.getDataRange().getValues();
  if (data.length === 0) return { ok: false, erro: 'Aba Clientes está vazia.' };

  var headers = data[0].map(function (h) { return String(h).trim(); });
  var colIdx  = {};
  headers.forEach(function (h, i) { colIdx[h] = i; });

  var idCol = colIdx['IdCliente'];
  if (idCol === undefined) return { ok: false, erro: 'Coluna IdCliente não encontrada na aba Clientes.' };

  var rowIdx = -1;
  for (var i = 1; i < data.length; i++) {
    if (String(data[i][idCol]) === String(idCliente)) { rowIdx = i + 1; break; }
  }
  if (rowIdx === -1) return { ok: false, erro: 'Cliente não encontrado (IdCliente ' + idCliente + ').' };

  // Sem checagem de dono aqui de propósito: essa action só é chamada pela
  // tela de Proposta (aba "Contrato"), que não tem login/sessão de usuário
  // — não existe um "solicitanteId" confiável pra conferir. Só edita
  // nome/CPF/endereço (nunca exclui nada), então o risco é baixo.

  var nomeCol = colIdx['Nome Razao Social'] !== undefined ? colIdx['Nome Razao Social'] : colIdx['Nome'];
  var cpfCol  = colIdx['CPF ou CNPJ']       !== undefined ? colIdx['CPF ou CNPJ']       : colIdx['CPF'];
  var endCol  = colIdx['Endereco'];
  var emailCol      = colIdx['Email'];
  var cpfEqCol      = colIdx['CPFEquatorial'];
  var dataNascEqCol = colIdx['DataNascimentoEquatorial'];

  if (body.nome !== undefined && body.nome !== null && nomeCol !== undefined) {
    sheet.getRange(rowIdx, nomeCol + 1).setValue(body.nome);
  }
  if (body.cpf !== undefined && body.cpf !== null && cpfCol !== undefined) {
    sheet.getRange(rowIdx, cpfCol + 1).setValue(body.cpf);
  }
  if (body.endereco !== undefined && body.endereco !== null && endCol !== undefined) {
    sheet.getRange(rowIdx, endCol + 1).setValue(body.endereco);
  }
  if (body.email !== undefined && body.email !== null && emailCol !== undefined) {
    sheet.getRange(rowIdx, emailCol + 1).setValue(body.email);
  }
  if (body.cpfEquatorial !== undefined && body.cpfEquatorial !== null && cpfEqCol !== undefined) {
    sheet.getRange(rowIdx, cpfEqCol + 1).setValue(body.cpfEquatorial);
  }
  if (body.dataNascimentoEquatorial !== undefined && body.dataNascimentoEquatorial !== null && dataNascEqCol !== undefined) {
    sheet.getRange(rowIdx, dataNascEqCol + 1).setValue(body.dataNascimentoEquatorial);
  }

  return {
    ok: true, acao: 'atualizado', idCliente: idCliente,
    nome: body.nome, cpf: body.cpf, endereco: body.endereco, email: body.email,
    cpfEquatorial: body.cpfEquatorial, dataNascimentoEquatorial: body.dataNascimentoEquatorial
  };
}

function saveProposal(body) {
  var sheet  = getOrCreatePropostasSheet();
  var values = sheet.getDataRange().getValues();

  var rowIdx = -1;
  var propostaId = body.id || generateShortId();
  if (body.id) {
    for (var i = 1; i < values.length; i++) {
      if (String(values[i][0]) === String(body.id)) {
        rowIdx = i + 1; break;
      }
    }
  }

  var baseUrl  = (body.baseUrl || 'https://pampfelp.github.io/relatorio-ponto/proposta.html').replace(/\/$/, '');
  var linkProposta = baseUrl + '?id=' + propostaId;

  var rowData = [
    propostaId,
    body.data         || formatDateBR(new Date()),
    body.numero       || '',
    body.idOportunidade || '',
    body.nomeCliente  || '',
    body.idVendedor   || '',
    body.nomeVendedor || '',
    parseFloat(body.totalServicos)      || 0,
    parseFloat(body.totalDeslocamento)  || 0,
    parseFloat(body.total)              || 0,
    JSON.stringify(body.blocos || []),
    body.observacoes  || '',
    body.status       || 'Rascunho',
    linkProposta
  ];

  if (rowIdx !== -1) {
    sheet.getRange(rowIdx, 1, 1, rowData.length).setValues([rowData]);
    return { ok: true, acao: 'atualizado', id: propostaId, link: linkProposta };
  } else {
    sheet.appendRow(rowData);
    var lastRow = sheet.getLastRow();
    var linkCol = 14; // coluna N = LinkProposta
    // Aspas duplicadas neutralizam qualquer tentativa de "fechar" a string da fórmula
    // via baseUrl (que vem do cliente) e injetar outra coisa dentro do HYPERLINK().
    sheet.getRange(lastRow, linkCol).setFormula('=HYPERLINK("' + linkProposta.replace(/"/g, '""') + '","Ver proposta")');
    return { ok: true, acao: 'criado', id: propostaId, link: linkProposta };
  }
}

function deleteProposal(body) {
  var id = body.id;
  if (!id) return { ok: false, erro: 'ID obrigatório.' };
  var sheet  = getSheetSafe(CONFIG.ABA_PROPOSTAS);
  if (!sheet) return { ok: false, erro: 'Aba Propostas não encontrada.' };
  var values = sheet.getDataRange().getValues();
  for (var i = 1; i < values.length; i++) {
    if (String(values[i][0]) === String(id)) {
      var dono = String(values[i][5] || ''); // coluna F = IdVendedor (mesma ordem gravada por saveProposal)
      var ehDono = dono && String(body.solicitanteId) === dono;
      if (!ehDono && !isSolicitanteAdmin(body.solicitanteId)) {

      }
      sheet.deleteRow(i + 1);
      return { ok: true, acao: 'deletado' };
    }
  }
  return { ok: false, erro: 'Proposta não encontrada.' };
}

// Colunas de autenticação que devem existir na aba Vendedores.
// São criadas automaticamente (no fim da planilha) na primeira vez que
// alguém tenta logar, pedir reset, etc — sem apagar nada que já existe.
var AUTH_COLUNAS = ['SenhaHash', 'SenhaSalt', 'ResetToken', 'ResetTokenExpira'];

// Funções (valores da coluna "Tipo") que têm acesso total ao sistema —
// veem todo mundo em Ponto/Vendas/Funil e podem administrar a aba Usuários.
// Mantenha esta lista em sincronia com SG_ADMIN_ROLES no index.html.
var ADMIN_ROLES = ['admin', 'administrador', 'ceo', 'gestor', 'gerente'];

/**
 * Confere se o IdVendedor informado corresponde a alguém com função administrativa.
 * Usado para proteger as ações de gerenciamento de usuários.
 */
function isSolicitanteAdmin(solicitanteId) {
  if (!solicitanteId) return false;
  var ctx = ensureVendedoresAuthColumns();
  var sheet = ctx.sheet, colIdx = ctx.colIdx;
  var idCol = colIdx['IdVendedor'], tipoCol = colIdx['Tipo'];
  if (!idCol || !tipoCol) return false;

  var values = sheet.getDataRange().getValues();
  for (var i = 1; i < values.length; i++) {
    if (String(values[i][idCol - 1]) === String(solicitanteId)) {
      var tipo = String(values[i][tipoCol - 1] || '').trim().toLowerCase();
      return ADMIN_ROLES.indexOf(tipo) !== -1;
    }
  }
  return false;
}

/**
 * Garante que a aba Vendedores tem as colunas de autenticação.
 * Retorna {sheet, colIdx} onde colIdx é um mapa NomeDaColuna -> número da coluna (1-based).
 */
function ensureVendedoresAuthColumns() {
  var sheet = getSheet(CONFIG.ABA_VENDEDORES);
  var lastCol = Math.max(sheet.getLastColumn(), 1);
  var headers = sheet.getRange(1, 1, 1, lastCol).getValues()[0].map(function (h) { return String(h).trim(); });

  var faltando = AUTH_COLUNAS.filter(function (h) { return headers.indexOf(h) === -1; });
  if (faltando.length) {
    sheet.getRange(1, lastCol + 1, 1, faltando.length).setValues([faltando]);
  }

  var novoLastCol = sheet.getLastColumn();
  headers = sheet.getRange(1, 1, 1, novoLastCol).getValues()[0].map(function (h) { return String(h).trim(); });
  var colIdx = {};
  headers.forEach(function (h, i) { if (h) colIdx[h] = i + 1; }); // 1-based

  return { sheet: sheet, colIdx: colIdx };
}

function sha256Hex(str) {
  var bytes = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, str, Utilities.Charset.UTF_8);
  return bytes.map(function (b) {
    b = (b < 0) ? b + 256 : b;
    return (b < 16 ? '0' : '') + b.toString(16);
  }).join('');
}

function gerarSalt() {
  return Utilities.getUuid().replace(/-/g, '');
}

function hashSenha(senha, salt) {
  return sha256Hex(AUTH_PEPPER + '|' + salt + '|' + senha);
}

function gerarTokenNumerico() {
  return String(Math.floor(100000 + Math.random() * 900000)); // 6 dígitos
}

/**
 * Localiza a linha do vendedor pelo e-mail (case-insensitive).
 * Retorna null se não encontrado, ou { rowIndex, values, colIdx, sheet }.
 */
function findVendedorRowByEmail(email) {
  var ctx = ensureVendedoresAuthColumns();
  var sheet = ctx.sheet, colIdx = ctx.colIdx;
  var emailCol = colIdx['Email'];
  if (!emailCol) return null;

  var values = sheet.getDataRange().getValues();
  var alvo = String(email || '').trim().toLowerCase();
  if (!alvo) return null;

  for (var i = 1; i < values.length; i++) {
    if (String(values[i][emailCol - 1] || '').trim().toLowerCase() === alvo) {
      return { rowIndex: i + 1, values: values[i], colIdx: colIdx, sheet: sheet };
    }
  }
  return null;
}

/**
 * action: login — body { email, senha }
 */
function actionLogin(body) {
  var email = body.email, senha = body.senha;
  if (!email || !senha) return { ok: false, erro: 'Informe e-mail e senha.' };

  var found = findVendedorRowByEmail(email);
  if (!found) return { ok: false, erro: 'E-mail não encontrado.' };

  var colIdx = found.colIdx, row = found.values;

  var statusCol = colIdx['Status'];
  if (statusCol && String(row[statusCol - 1] || '').trim().toLowerCase() === 'inativo') {
    return { ok: false, erro: 'Usuário inativo. Fale com o administrador.' };
  }

  var hashCol = colIdx['SenhaHash'], saltCol = colIdx['SenhaSalt'];
  var hashArmazenado = String(row[hashCol - 1] || '');
  var salt = String(row[saltCol - 1] || '');

  var usuario = {
    idVendedor: row[colIdx['IdVendedor'] - 1],
    nome:       row[colIdx['Nome'] - 1],
    email:      row[colIdx['Email'] - 1],
    tipo:       colIdx['Tipo'] ? row[colIdx['Tipo'] - 1] : ''
  };

  if (!hashArmazenado) {
    // Ainda não tem senha própria: só entra com a senha padrão, e é
    // obrigado a trocar por uma senha só dele antes de acessar o sistema.
    if (senha !== SENHA_PADRAO) {
      return {
        ok: false, erro: 'SEM_SENHA',
        mensagem: 'Você ainda não tem uma senha própria. Use a senha padrão informada pela empresa para o primeiro acesso.'
      };
    }
    return { ok: true, usuario: usuario, precisaTrocarSenha: true };
  }

  if (hashSenha(senha, salt) !== hashArmazenado) {
    return { ok: false, erro: 'Senha incorreta.' };
  }

  return { ok: true, usuario: usuario };
}

/**
 * action: esqueciSenha — body { email }
 * Gera um código de 6 dígitos, salva na planilha com validade de 30 min
 * e envia por e-mail via MailApp (usa a conta Google dona do Apps Script).
 * Sempre responde ok:true (não revela se o e-mail existe, por segurança).
 */
function actionEsqueciSenha(body) {
  var email = body.email;
  if (!email) return { ok: false, erro: 'Informe o e-mail.' };

  var found = findVendedorRowByEmail(email);
  if (!found) return { ok: true }; // não revela se o e-mail existe

  var colIdx = found.colIdx, sheet = found.sheet, rowIndex = found.rowIndex;
  var token  = gerarTokenNumerico();
  var expira = new Date(Date.now() + 30 * 60 * 1000); // 30 minutos

  sheet.getRange(rowIndex, colIdx['ResetToken']).setValue(token);
  sheet.getRange(rowIndex, colIdx['ResetTokenExpira']).setValue(expira);

  var nome = found.values[colIdx['Nome'] - 1] || '';

  try {
    MailApp.sendEmail({
      to: email,
      subject: 'Solar Green · Código para redefinir sua senha',
      body:
        'Olá ' + nome + ',\n\n' +
        'Recebemos um pedido para redefinir a sua senha no sistema Solar Green.\n' +
        'Seu código de verificação é: ' + token + '\n\n' +
        'Esse código expira em 30 minutos. Se você não solicitou isso, pode ignorar este e-mail.\n\n' +
        '— Solar Green Suporte'
    });
  } catch (err) {
    return { ok: false, erro: 'Não foi possível enviar o e-mail: ' + err.message };
  }

  return { ok: true };
}

/**
 * action: redefinirSenha — body { email, token, novaSenha }
 * Usado no fluxo "esqueci minha senha" (não exige senha atual).
 */
function actionRedefinirSenha(body) {
  var email = body.email, token = body.token, novaSenha = body.novaSenha;
  if (!email || !token || !novaSenha) return { ok: false, erro: 'Preencha todos os campos.' };
  if (String(novaSenha).length < 6) return { ok: false, erro: 'A senha deve ter pelo menos 6 caracteres.' };

  var found = findVendedorRowByEmail(email);
  if (!found) return { ok: false, erro: 'E-mail não encontrado.' };

  var colIdx = found.colIdx, sheet = found.sheet, rowIndex = found.rowIndex, row = found.values;

  var tokenArmazenado = String(row[colIdx['ResetToken'] - 1] || '');
  var expiraVal = row[colIdx['ResetTokenExpira'] - 1];

  if (!tokenArmazenado || tokenArmazenado !== String(token).trim()) {
    return { ok: false, erro: 'Código inválido.' };
  }

  var expiraDate = (expiraVal instanceof Date) ? expiraVal : new Date(expiraVal);
  if (!expiraDate || isNaN(expiraDate.getTime()) || expiraDate.getTime() < Date.now()) {
    return { ok: false, erro: 'Código expirado. Solicite um novo.' };
  }

  var novoSalt = gerarSalt();
  var novoHash = hashSenha(novaSenha, novoSalt);

  sheet.getRange(rowIndex, colIdx['SenhaHash']).setValue(novoHash);
  sheet.getRange(rowIndex, colIdx['SenhaSalt']).setValue(novoSalt);
  sheet.getRange(rowIndex, colIdx['ResetToken']).setValue('');
  sheet.getRange(rowIndex, colIdx['ResetTokenExpira']).setValue('');

  return { ok: true };
}

/**
 * action: trocarSenha — body { idVendedor, senhaAtual, novaSenha }
 * Usado quando o usuário já está logado e quer trocar a própria senha
 * (não precisa de e-mail nem de token).
 */
function actionTrocarSenha(body) {
  var idVendedor = body.idVendedor, senhaAtual = body.senhaAtual, novaSenha = body.novaSenha;
  if (!idVendedor || !senhaAtual || !novaSenha) return { ok: false, erro: 'Preencha todos os campos.' };
  if (String(novaSenha).length < 6) return { ok: false, erro: 'A nova senha deve ter pelo menos 6 caracteres.' };

  var ctx = ensureVendedoresAuthColumns();
  var sheet = ctx.sheet, colIdx = ctx.colIdx;
  var values = sheet.getDataRange().getValues();
  var idCol = colIdx['IdVendedor'];

  var rowIndex = -1, row = null;
  for (var i = 1; i < values.length; i++) {
    if (String(values[i][idCol - 1]) === String(idVendedor)) { rowIndex = i + 1; row = values[i]; break; }
  }
  if (rowIndex === -1) return { ok: false, erro: 'Usuário não encontrado.' };

  var hashArmazenado = String(row[colIdx['SenhaHash'] - 1] || '');
  var salt = String(row[colIdx['SenhaSalt'] - 1] || '');

  if (hashArmazenado && hashSenha(senhaAtual, salt) !== hashArmazenado) {
    return { ok: false, erro: 'Senha atual incorreta.' };
  }

  var novoSalt = gerarSalt();
  var novoHash = hashSenha(novaSenha, novoSalt);
  sheet.getRange(rowIndex, colIdx['SenhaHash']).setValue(novoHash);
  sheet.getRange(rowIndex, colIdx['SenhaSalt']).setValue(novoSalt);

  return { ok: true };
}

/**
 * action: listVendedoresAdmin — body { solicitanteId }
 * Retorna a lista completa de vendedores (sem os campos sensíveis de senha),
 * só pra quem tem função administrativa.
 */
function actionListVendedoresAdmin(body) {

  }

  var ctx = ensureVendedoresAuthColumns();
  var sheet = ctx.sheet, colIdx = ctx.colIdx;
  var camposOcultos = { SenhaHash: 1, SenhaSalt: 1, ResetToken: 1, ResetTokenExpira: 1 };
  var headers = Object.keys(colIdx);

  var lista = [];
  for (var i = 1; i < values.length; i++) {

    if (row.every(function (c) { return c === '' || c === null; })) continue;
    var obj = {};
    headers.forEach(function (h) {
      if (camposOcultos[h]) return;
      var val = row[colIdx[h] - 1];
      if (isDateLike(val)) val = formatDateBR(val);

    });

/**
 * action: salvarVendedor — body { solicitanteId, idVendedor?, nome, email, telefone?, tipo?, status? }
 * Se idVendedor vier preenchido, atualiza a linha existente; senão, cria um vendedor novo.
 * Nunca mexe nas colunas de senha — isso só acontece por login/trocarSenha/redefinirSenha.
 */
function actionSalvarVendedor(body) {
  if (!isSolicitanteAdmin(body.solicitanteId)) {
    return { ok: false, erro: 'Acesso restrito a administradores.' };
  }

  var nome = (body.nome || '').trim(), email = (body.email || '').trim();
  if (!nome || !email) return { ok: false, erro: 'Nome e e-mail são obrigatórios.' };

  var ctx = ensureVendedoresAuthColumns();
  var sheet = ctx.sheet, colIdx = ctx.colIdx;
  var idCol = colIdx['IdVendedor'];

  // Se for edição, confere se o novo e-mail já não pertence a OUTRO vendedor.
  var values = sheet.getDataRange().getValues();
  var emailCol = colIdx['Email'];
  for (var k = 1; k < values.length; k++) {
    var mesmoEmail = emailCol && String(values[k][emailCol - 1] || '').trim().toLowerCase() === email.toLowerCase();
    var linhaDiferente = !body.idVendedor || String(values[k][idCol - 1]) !== String(body.idVendedor);
    if (mesmoEmail && linhaDiferente) {
      return { ok: false, erro: 'Já existe outro usuário com esse e-mail.' };
    }
  }

  var rowIndex = -1;
  if (body.idVendedor) {
    for (var i = 1; i < values.length; i++) {
      if (String(values[i][idCol - 1]) === String(body.idVendedor)) { rowIndex = i + 1; break; }
    }
  }

  var campos = {
    'Nome': nome,
    'Email': email
  };
  if (body.telefone !== undefined) campos['Telefone'] = body.telefone;
  if (body.tipo     !== undefined) campos['Tipo']     = body.tipo;
  if (body.status   !== undefined) campos['Status']   = body.status;

  if (rowIndex === -1) {
    var novoId = body.idVendedor || generateShortId(); // usa o ID que o cliente já gerou (otimista) — só gera aqui se não veio
    var lastCol = sheet.getLastColumn();
    var novaLinha = new Array(lastCol).fill('');
    Object.keys(colIdx).forEach(function (h) {
      if (h === 'IdVendedor') novaLinha[colIdx[h] - 1] = novoId;
      else if (campos[h] !== undefined) novaLinha[colIdx[h] - 1] = campos[h];
    });
    sheet.getRange(sheet.getLastRow() + 1, 1, 1, novaLinha.length).setValues([novaLinha]);
    invalidarCache('sh_vendedores');
    return { ok: true, acao: 'criado', idVendedor: novoId };
  } else {
    Object.keys(campos).forEach(function (h) {
      if (!colIdx[h]) return;
      sheet.getRange(rowIndex, colIdx[h]).setValue(campos[h]);
    });
    invalidarCache('sh_vendedores');
    return { ok: true, acao: 'atualizado', idVendedor: body.idVendedor };
  }
}

// Pasta de dados do app no Drive — é onde o AppSheet realmente guarda as fotos
// (caminho completo: Meu Drive / appsheet / data / CRMResolve-360750226 / <pasta da tabela> / <arquivo>).
// Os caminhos salvos na planilha só têm a parte final (ex: "Agendamentos Respostas_Images/xxx.png"),
// relativa a essa pasta — por isso a busca direta pelo nome nem sempre achava o arquivo certo.
var APPSHEET_PASTA_APP = 'CRMResolve-360750226';

function encontrarPastaPorNome(nome, pastaBase) {
  var it = pastaBase ? pastaBase.getFoldersByName(nome) : DriveApp.getFoldersByName(nome);
  return it.hasNext() ? it.next() : null;
}

/**
 * As fotos respondidas pelo AppSheet (antes do sistema novo) foram salvas como
 * um CAMINHO relativo dentro do Drive (ex: "Agendamentos Respostas_Images/xxx.png"),
 * relativo à pasta de dados do próprio app — não uma URL. Essa função resolve isso:
 * procura o arquivo primeiro dentro da pasta do app, depois em qualquer pasta com
 * esse nome, e por último em todo o Drive; gera o link público e REGRAVA a célula
 * com a URL — assim isso só roda uma vez por foto antiga.
 */
function resolverEMigrarFotoLegada(sheet, colIdx, rowIndex, valorAtual) {
  if (!valorAtual) return valorAtual;
  var valor = String(valorAtual).trim();

  // Link antigo no formato "uc?id=..." — troca pro formato "thumbnail?id=..." que
  // carrega direto como imagem com mais confiabilidade (e regrava a célula).
  var mUc = valor.match(/drive\.google\.com\/uc\?id=([\w-]+)/);
  if (mUc) {
    try {
      var arquivoAntigo = DriveApp.getFileById(mUc[1]);
      arquivoAntigo.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    } catch (e) {
      // se não conseguir nem achar o arquivo pelo ID, segue o baile — só troca o formato do link mesmo
    }
    var urlNova = 'https://drive.google.com/thumbnail?id=' + mUc[1] + '&sz=w2000';
    if (colIdx['RespostaFoto']) sheet.getRange(rowIndex, colIdx['RespostaFoto']).setValue(urlNova);
    return urlNova;
  }

  if (/^https?:\/\//.test(valor)) return valor; // já é uma URL válida (outro formato conhecido)

  try {
    var partes = valor.split('/');
    var nomeArquivo = partes.pop();
    var nomePasta = partes.length ? partes.join('/') : null; // ex: "Agendamentos Respostas_Images"

    var arquivo = null;

    // 1) Dentro da pasta de dados do app (onde o AppSheet realmente guarda as fotos)
    var pastaApp = encontrarPastaPorNome(APPSHEET_PASTA_APP);
    if (pastaApp) {
      var pastaAlvo = nomePasta ? encontrarPastaPorNome(nomePasta, pastaApp) : pastaApp;
      if (pastaAlvo) {
        var it1 = pastaAlvo.getFilesByName(nomeArquivo);
        if (it1.hasNext()) arquivo = it1.next();
      }
    }

    // 2) Qualquer pasta com esse nome, em qualquer lugar do Drive
    if (!arquivo && nomePasta) {
      var pasta2 = encontrarPastaPorNome(nomePasta);
      if (pasta2) {
        var it2 = pasta2.getFilesByName(nomeArquivo);
        if (it2.hasNext()) arquivo = it2.next();
      }
    }

    // 3) Último recurso: procura o arquivo pelo nome em todo o Drive
    if (!arquivo) {
      var it3 = DriveApp.getFilesByName(nomeArquivo);
      if (it3.hasNext()) arquivo = it3.next();
    }

    if (!arquivo) return valor; // não achou de jeito nenhum — mantém como estava

    try { arquivo.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); } catch (e) {}
    var url = 'https://drive.google.com/thumbnail?id=' + arquivo.getId() + '&sz=w2000';
    if (colIdx['RespostaFoto']) sheet.getRange(rowIndex, colIdx['RespostaFoto']).setValue(url);
    return url;
  } catch (err) {
    return valor; // se der qualquer erro na busca, não trava a tela — só não migra essa foto agora
  }
}

/**
 * Lê a aba Agendamentos Respostas migrando de passagem qualquer foto antiga
 * (caminho relativo do AppSheet) para uma URL de verdade.
 */
function lerAgendamentosRespostasComMigracao(filtroIdsAgendamentos) {
  var LIMITE_RESOLUCOES_POR_CHAMADA = 40; // já é por página (10 agendamentos), não a aba inteira — pode ser mais generoso
  var sheet = getOrCreateAgendamentosRespostasSheet();
  var values = sheet.getDataRange().getValues();
  if (values.length < 2) return [];
  var headers = values[0].map(function (h) { return String(h).trim(); });
  var colIdx = {};
  headers.forEach(function (h, i) { if (h) colIdx[h] = i + 1; });
  var idAgCol = colIdx['IdAgendamento'];

  var out = [];
  var resolucoesFeitas = 0;
  for (var i = 1; i < values.length; i++) {
    var row = values[i];
    if (row.every(function (c) { return c === '' || c === null; })) continue;
    if (filtroIdsAgendamentos && idAgCol && !filtroIdsAgendamentos[row[idAgCol - 1]]) continue;
    var obj = {};
    headers.forEach(function (h, j) { if (h) obj[h] = row[j]; });
    var fotoAtual = colIdx['RespostaFoto'] ? String(obj['RespostaFoto'] || '') : '';
    var precisaMigrar = fotoAtual && !/^https?:\/\//.test(fotoAtual);
    var linkAntigoUc = fotoAtual && /drive\.google\.com\/uc\?id=/.test(fotoAtual);
    if (precisaMigrar || linkAntigoUc) {
      // Link "uc?id=" é só troca de formato (rápido, não conta na cota); caminho
      // relativo antigo do AppSheet precisa procurar no Drive (mais lento, tem cota).
      if (linkAntigoUc || resolucoesFeitas < LIMITE_RESOLUCOES_POR_CHAMADA) {
        if (precisaMigrar) resolucoesFeitas++;
        obj['RespostaFoto'] = resolverEMigrarFotoLegada(sheet, colIdx, i + 1, obj['RespostaFoto']);
      }
      // senão: deixa o caminho antigo como está por agora — o front-end mostra o aviso de
      // "link não encontrado" e essa foto é resolvida numa próxima vez que a tela recarregar.
    }
    out.push(obj);
  }
  return out;
}

function parseDataISO(s) {
  var m = String(s || '').trim().match(/^(\d{4})-(\d{2})-(\d{2})/);
  if (!m) return null;
  return new Date(parseInt(m[1], 10), parseInt(m[2], 10) - 1, parseInt(m[3], 10));
}

function findAgendamentoRowById(id) {
  if (!id) return null;
  var sheet  = getOrCreateAgendamentosSheet();
  var colIdx = agendamentosColIdx(sheet);
  var idCol  = colIdx['IdAgendamento'];
  if (!idCol) return null;
  var values = sheet.getDataRange().getValues();
  for (var i = 1; i < values.length; i++) {
    if (String(values[i][idCol - 1]) === String(id)) {
      return { sheet: sheet, colIdx: colIdx, rowIndex: i + 1, values: values[i] };
    }
  }
  return null;
}

/**
 * Um solicitante pode mexer num agendamento se for administrador
 * OU se for o técnico responsável por aquele agendamento específico.
 */
function isTecnicoOuAdmin(solicitanteId, idAgendamento) {
  if (!solicitanteId) return false;
  if (isSolicitanteAdmin(solicitanteId)) return true;
  var found = findAgendamentoRowById(idAgendamento);
  if (!found) return false;
  var tecnicoCol = found.colIdx['TecnicoResponsavel'];
  if (!tecnicoCol) return false;
  return String(found.values[tecnicoCol - 1]) === String(solicitanteId);
}

/**
 * Um solicitante pode gerar/enviar a Ordem de Serviço pra assinatura se for
 * administrador OU se for o vendedor responsável pelo CLIENTE daquele
 * agendamento — quem manda a OS pra assinatura é o vendedor (dono da
 * negociação), não o técnico que executou o serviço em campo (esse já tem
 * sua própria regra em isTecnicoOuAdmin, usada só pro upload de fotos).
 *
 * 2026-09-16: depois da migração pra Firestore, nem o solicitante (agora um
 * uid do Firebase Auth) nem o agendamento em si existem mais na planilha —
 * a checagem antiga (isSolicitanteAdmin + findAgendamentoRowById, ambos
 * batendo na aba Vendedores/Agendamentos) sempre negava, mesmo pra admin.
 * Tentei uma ponte via REST do Firestore usando o idToken do usuário como
 * Bearer, mas a API pública do Firestore não aceita idToken do Firebase
 * Auth desse jeito (só o SDK do navegador fala com o Firestore por esse
 * canal) — caía numa cota anônima e estourava em 429 "Quota exceeded".
 * Solução: o navegador (que já leu esse Firestore certinho pra montar a
 * própria tela) manda "souAdmin" e "vendedorResponsavelCliente" já
 * resolvidos. Isso não abre brecha nova — firestore.rules nessa fase só
 * exige "isSignedIn()" (sem checar dono), então qualquer usuário logado já
 * conseguia ler esses mesmos dados direto pelo SDK; essa checagem sempre
 * foi regra de negócio (só o vendedor certo aciona isso), não fronteira de
 * segurança. Sem esses 2 campos (chamada antiga), cai no caminho velho via
 * planilha, sem quebrar nada que já funcionava antes da migração.
 */
function isVendedorDoClienteOuAdmin_(solicitanteId, idAgendamento, souAdmin, vendedorResponsavelCliente) {
  if (!solicitanteId) return false;
  if (souAdmin || isSolicitanteAdmin(solicitanteId)) return true;
  if (vendedorResponsavelCliente !== undefined && vendedorResponsavelCliente !== null) {
    return String(vendedorResponsavelCliente || '') === String(solicitanteId);
  }
  var found = findAgendamentoRowById(idAgendamento);
  if (!found) return false;
  var clienteCol = found.colIdx['IdCliente'];
  if (!clienteCol) return false;
  var idCliente = found.values[clienteCol - 1];
  var clientes = sheetToObjects(getSheet(CONFIG.ABA_CLIENTES));
  var clienteObj = clientes.filter(function (c) { return String(c.IdCliente) === String(idCliente); })[0];
  if (!clienteObj) return false;
  return String(clienteObj['Vendedor Responsavel'] || '') === String(solicitanteId);
}

function minutosDeHoraGS(hhmm) {
  var m = String(hhmm || '').match(/^(\d{1,2}):(\d{2})/);
  if (!m) return null;
  return parseInt(m[1], 10) * 60 + parseInt(m[2], 10);
}

function normalizarTextoGS(s) {
  return String(s || '').trim().toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '');
}

function formatarDataBRGS(iso) {
  var m = String(iso || '').match(/^(\d{4})-(\d{2})-(\d{2})/);
  return m ? (m[3] + '/' + m[2] + '/' + m[1]) : String(iso || '');
}

/**
 * action: salvarAgendamento — body { idAgendamento?, idCliente, idContrato?, idServico,
 *   dataInicio ('YYYY-MM-DD'), horaInicio, horaFim, tecnicoResponsavel, statusAgendamento?,
 *   observacaoComercial?, quantidadeModulos?, modeloModulos?, quantidadeInversores?, modeloInversores? }
 * Criado por vendedor/admin — não exige que o solicitante seja o técnico.
 *
 * Réplica no servidor das mesmas regras da tela (que por sua vez replicam as
 * regras que já existiam no AppSheet): cliente precisa ter endereço, hora fim
 * > hora início, e sem conflito de horário pro mesmo técnico no mesmo dia.
 * Isso garante que ninguém consiga burlar a validação chamando a API direto.
 */
function actionSalvarAgendamento(body) {
  var idCliente = body.idCliente, idServico = body.idServico,
      dataInicio = body.dataInicio, tecnico = body.tecnicoResponsavel;
  if (!idCliente || !idServico || !dataInicio || !tecnico) {
    return { ok: false, erro: 'Cliente, serviço, data de início e técnico responsável são obrigatórios.' };
  }
  // Regra pedida pelo Felipe: a OS não pode sair sem valor definido — evita
  // agendamento (e depois proposta/cobrança) sem preço registrado.
  var valorNum = paraNumero(body.valor);
  if (valorNum === '' || valorNum <= 0) {
    return { ok: false, erro: 'Informe o valor do serviço.' };
  }
  if (!body.horaInicio || !body.horaFim) {
    return { ok: false, erro: 'Escolha o horário de início e fim.' };
  }
  var statusNovo = body.statusAgendamento || 'Agendado';
  var canceladoNovo = normalizarTextoGS(statusNovo) === 'cancelado';
  if (canceladoNovo && !String(body.motivoCancelamento || '').trim()) {
    return { ok: false, erro: 'Informe o motivo do cancelamento.' };
  }

  // Regra 1: cliente precisa ter endereço cadastrado antes de agendar.
  var clientes = sheetToObjects(getSheet(CONFIG.ABA_CLIENTES));
  var clienteObj = clientes.filter(function (c) { return String(c.IdCliente) === String(idCliente); })[0];
  if (!clienteObj || !String((clienteObj.Endereco || clienteObj['Endereço']) || '').trim()) {
    return { ok: false, erro: '⚠️ Este cliente não possui endereço cadastrado. Atualize o cadastro do cliente antes de realizar o agendamento.' };
  }

  // Regra 1b: cliente precisa ter e-mail cadastrado antes de agendar — é pra
  // lá que vai o link de assinatura digital da Ordem de Serviço depois do
  // atendimento; sem e-mail, o cliente nunca recebe o link pra assinar.
  if (!String(clienteObj.Email || '').trim()) {
    return { ok: false, erro: '⚠️ Este cliente não possui e-mail cadastrado. Atualize o cadastro do cliente antes de realizar o agendamento.' };
  }

  // Regra 2: hora fim tem que ser maior que hora início.
  var minIni = minutosDeHoraGS(body.horaInicio), minFim = minutosDeHoraGS(body.horaFim);
  if (minIni === null || minFim === null || minFim <= minIni) {
    return { ok: false, erro: 'O horário fim deve ser maior que o início.' };
  }

  // Regra 3: sem conflito de horário pro mesmo técnico no mesmo dia (cancelados não contam).
  // O check + a gravação ficam dentro de um LockService: sem isso, duas requisições
  // simultâneas podem ler a planilha ANTES de qualquer uma escrever, nenhuma vê o
  // agendamento da outra, e as duas passam no check — dois agendamentos conflitantes
  // seriam salvos com sucesso ao mesmo tempo (é exatamente esse buraco que o
  // comentário da função já dizia que não devia existir).
  var lock = LockService.getScriptLock();
  try {
    lock.waitLock(10000);
  } catch (e) {
    return { ok: false, erro: 'Sistema ocupado, tente novamente em alguns segundos.' };
  }
  try {
    var todosAgendamentos = sheetToObjects(getOrCreateAgendamentosSheet());
    var conflito = todosAgendamentos.filter(function (a) {
      if (body.idAgendamento && String(a.IdAgendamento) === String(body.idAgendamento)) return false;
      if (String(a.TecnicoResponsavel) !== String(tecnico)) return false;
      if (normalizarTextoGS(a['Status Agendamento']) === 'cancelado') return false;
      if (normalizeDateKey(a['Data Inicio']) !== dataInicio) return false;
      var aIni = minutosDeHoraGS(a['Hora inicio']), aFim = minutosDeHoraGS(a['Hora Fim']);
      if (aIni === null || aFim === null) return false;
      return aIni < minFim && aFim > minIni;
    })[0];
    if (conflito) {
      var vendedores = getSheetObjectsComCache(CONFIG.ABA_VENDEDORES, 'sh_vendedores');
      var tecnicoObj = vendedores.filter(function (v) { return String(v.IdVendedor) === String(tecnico); })[0];
      var nomeTecnico = tecnicoObj ? tecnicoObj.Nome : tecnico;
      return { ok: false, erro: 'Conflito! O técnico ' + nomeTecnico + ' já possui agendamento neste horário no dia ' + formatarDataBRGS(dataInicio) + '.' };
    }

    var sheet  = getOrCreateAgendamentosSheet();
    var colIdx = agendamentosColIdx(sheet);
    var idCol  = colIdx['IdAgendamento'];

    var rowIndex = -1;
    if (body.idAgendamento) {
      var values = sheet.getDataRange().getValues();
      for (var i = 1; i < values.length; i++) {
        if (String(values[i][idCol - 1]) === String(body.idAgendamento)) { rowIndex = i + 1; break; }
      }
    }

    var campos = {
      'IdCliente': idCliente,
      'IdContrato': body.idContrato || '',
      'IdServico': idServico,
      'Valor': valorNum,
      'Data Inicio': parseDataISO(dataInicio),
      'Hora inicio': body.horaInicio || '',
      'Hora Fim': body.horaFim || '',
      'TecnicoResponsavel': tecnico,
      'Status Agendamento': statusNovo,
      'Motivo Cancelamento': canceladoNovo ? body.motivoCancelamento : '',
      'Observacao Comercial': body.observacaoComercial || '',
      'Quantidade de Modulos': body.quantidadeModulos !== undefined ? body.quantidadeModulos : '',
      'Modelo Modulos': body.modeloModulos || '',
      'Quantidade Inversores': body.quantidadeInversores !== undefined ? body.quantidadeInversores : '',
      'Modelo Inversores': body.modeloInversores || ''
    };

    if (rowIndex === -1) {
      var novoId = body.idAgendamento || generateShortId(); // usa o ID que o cliente já gerou (otimista) — só gera aqui se não veio
      var lastCol = sheet.getLastColumn();
      var novaLinha = new Array(lastCol).fill('');
      Object.keys(colIdx).forEach(function (h) {
        if (h === 'IdAgendamento') novaLinha[colIdx[h] - 1] = novoId;
        else if (campos[h] !== undefined) novaLinha[colIdx[h] - 1] = campos[h];
      });
      sheet.getRange(sheet.getLastRow() + 1, 1, 1, novaLinha.length).setValues([novaLinha]);
      return { ok: true, acao: 'criado', idAgendamento: novoId };
    } else {
      Object.keys(campos).forEach(function (h) {
        if (!colIdx[h]) return;
        sheet.getRange(rowIndex, colIdx[h]).setValue(campos[h]);
      });
      return { ok: true, acao: 'atualizado', idAgendamento: body.idAgendamento };
    }
  } finally {
    lock.releaseLock();
  }
}

/**
 * action: excluirAgendamento — body { idAgendamento }
 * Remove a linha do agendamento da aba Agendamentos. As respostas já
 * preenchidas pelo técnico (aba Agendamentos Respostas) não são apagadas —
 * ficam órfãs, mas isso não afeta nada (não são exibidas em lugar nenhum
 * sem o agendamento existir).
 */
function actionExcluirAgendamento(body) {
  if (!isSolicitanteAdmin(body.solicitanteId)) {
    return { ok: false, erro: 'Acesso restrito a administradores.' };
  }
  var idAgendamento = body.idAgendamento;
  if (!idAgendamento) return { ok: false, erro: 'idAgendamento é obrigatório.' };
  var sheet = getOrCreateAgendamentosSheet();
  var colIdx = agendamentosColIdx(sheet);
  var idCol = colIdx['IdAgendamento'];
  if (!idCol) return { ok: false, erro: 'Coluna IdAgendamento não encontrada.' };
  var values = sheet.getDataRange().getValues();
  for (var i = 1; i < values.length; i++) {
    if (String(values[i][idCol - 1]) === String(idAgendamento)) {
      sheet.deleteRow(i + 1);
      return { ok: true };
    }
  }
  return { ok: false, erro: 'Agendamento não encontrado.' };
}

  var vendedores  = getSheetObjectsComCache(CONFIG.ABA_VENDEDORES, 'sh_vendedores');

    servicos: servicos, templates: templates,
    atualizadoEm: new Date().toISOString()
  };
}

/**
 * action: getRespostasAgendamentos — body { solicitanteId, idsAgendamentos:[...] }
 * Busca as respostas SÓ dos agendamentos pedidos (ex: os 10 da página atual).
 * É essa filtragem por ID que evita varrer/migrar a aba inteira a cada carregamento.
 */
function actionGetRespostasAgendamentos(body) {
  var solicitanteId = body.solicitanteId;
  var ids = body.idsAgendamentos || [];
  if (!solicitanteId || !ids.length) return { ok: true, respostas: [] };

  var admin = isSolicitanteAdmin(solicitanteId);
  var idsPermitidos = {};
  ids.forEach(function (id) {
    if (admin || isTecnicoOuAdmin(solicitanteId, id)) idsPermitidos[id] = true;
  });
  if (!Object.keys(idsPermitidos).length) return { ok: true, respostas: [] };

  var respostas = lerAgendamentosRespostasComMigracao(idsPermitidos);
  return { ok: true, respostas: respostas };
}

/**
 * action: diagnosticarFotos — sem parâmetros.
 * Testa, passo a passo, cada parte do fluxo de fotos (achar a pasta do app,
 * resolver até 3 fotos antigas de exemplo, criar/compartilhar/apagar um
 * arquivo de teste na pasta de fotos novas) e devolve um relatório detalhado
 * de cada etapa — pra diagnosticar sem precisar ficar adivinhando.
 */
function actionDiagnosticarFotos() {
  var testes = [];

  try {
    var pastaApp = encontrarPastaPorNome(APPSHEET_PASTA_APP);
    testes.push({ passo: 'Achar pasta "' + APPSHEET_PASTA_APP + '" no Drive', ok: !!pastaApp, detalhe: pastaApp ? ('encontrada, id=' + pastaApp.getId()) : 'NÃO encontrada' });
  } catch (e) {
    testes.push({ passo: 'Achar pasta "' + APPSHEET_PASTA_APP + '" no Drive', ok: false, detalhe: 'ERRO: ' + e.message });
  }

  try {
    var sheet = getOrCreateAgendamentosRespostasSheet();
    var values = sheet.getDataRange().getValues();
    var headers = values[0].map(function (h) { return String(h).trim(); });
    var colIdx = {};
    headers.forEach(function (h, i) { colIdx[h] = i + 1; });

    var exemplos = [];
    for (var i = 1; i < values.length && exemplos.length < 3; i++) {
      var v = colIdx['RespostaFoto'] ? values[i][colIdx['RespostaFoto'] - 1] : '';
      if (v && !/^https?:\/\//.test(String(v))) exemplos.push({ linha: i + 1, valor: String(v) });
    }
    if (!exemplos.length) {
      testes.push({ passo: 'Procurar fotos antigas de exemplo pra testar', ok: true, detalhe: 'Nenhuma foto pendente de migração encontrada (ou já foram todas migradas).' });
    }
    exemplos.forEach(function (ex) {
      try {
        var resolvida = resolverEMigrarFotoLegada(sheet, colIdx, ex.linha, ex.valor);
        testes.push({ passo: 'Resolver "' + ex.valor + '"', ok: /^https?:\/\//.test(resolvida), detalhe: resolvida });
      } catch (e) {
        testes.push({ passo: 'Resolver "' + ex.valor + '"', ok: false, detalhe: 'ERRO: ' + e.message });
      }
    });
  } catch (e) {
    testes.push({ passo: 'Ler aba Agendamentos Respostas', ok: false, detalhe: 'ERRO: ' + e.message });
  }

  try {
    var pastaFotos = getOrCreatePastaFotosAgendamentos();
    var arquivoTeste = pastaFotos.createFile(Utilities.newBlob('teste de diagnóstico', 'text/plain', 'teste_diagnostico.txt'));
    arquivoTeste.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
    arquivoTeste.setTrashed(true);
    testes.push({ passo: 'Criar + compartilhar + apagar arquivo de teste (fluxo de upload novo)', ok: true, detalhe: 'OK' });
  } catch (e) {
    testes.push({ passo: 'Criar + compartilhar + apagar arquivo de teste (fluxo de upload novo)', ok: false, detalhe: 'ERRO: ' + e.message });
  }

  var tudoOk = testes.every(function (t) { return t.ok; });
  return { ok: true, tudoOk: tudoOk, testes: testes };
}

/**
 * action: atualizarStatusAgendamento — body { solicitanteId, idAgendamento, status }
 * Usado pelo técnico pra marcar "Em Andamento" / "Concluído" / "Cancelado".
 */
function actionAtualizarStatusAgendamento(body) {
  var idAgendamento = body.idAgendamento, status = body.status;
  if (!idAgendamento || !status) return { ok: false, erro: 'Dados incompletos.' };
  if (!isTecnicoOuAdmin(body.solicitanteId, idAgendamento)) {
    return { ok: false, erro: 'Sem permissão para essa ordem de serviço.' };
  }
  var statusNorm = normalizarTextoGS(status);
  if (statusNorm === 'cancelado' && !String(body.motivoCancelamento || '').trim()) {
    return { ok: false, erro: 'Informe o motivo do cancelamento.' };
  }

  var found = findAgendamentoRowById(idAgendamento);
  if (!found) return { ok: false, erro: 'Agendamento não encontrado.' };

  found.sheet.getRange(found.rowIndex, found.colIdx['Status Agendamento']).setValue(status);
  if (found.colIdx['Motivo Cancelamento']) {
    found.sheet.getRange(found.rowIndex, found.colIdx['Motivo Cancelamento']).setValue(statusNorm === 'cancelado' ? body.motivoCancelamento : '');
  }

  if (statusNorm === 'concluido' && found.colIdx['DataRelatorio']) {
    found.sheet.getRange(found.rowIndex, found.colIdx['DataRelatorio']).setValue(new Date());
  }

  return { ok: true };
}

/**
 * Cria ou atualiza a resposta de um campo do template pra um agendamento
 * (upsert por IdAgendamento + IdTemplate). "campo" é qual coluna recebe o
 * valor: 'RespostaTexto', 'RespostaFoto' ou 'RespostaQuantidade' — são as
 * 3 colunas reais da aba (a mesma linha de resposta usa só uma delas).
 */
function salvarOuAtualizarResposta(idAgendamento, idTemplate, campo, valor) {
  var sheet  = getOrCreateAgendamentosRespostasSheet();
  var values = sheet.getDataRange().getValues();
  var headers = values.length ? values[0].map(function (h) { return String(h).trim(); }) : AGENDAMENTOS_RESPOSTAS_HEADERS.slice();
  var colIdx = {};
  headers.forEach(function (h, i) { colIdx[h] = i + 1; });
  if (!colIdx[campo]) return; // coluna não existe nessa planilha — não faz nada, evita erro

  var idAgCol = colIdx['IdAgendamento'], idTplCol = colIdx['IdTemplate'];
  for (var i = 1; i < values.length; i++) {
    if (String(values[i][idAgCol - 1]) === String(idAgendamento) &&
        String(values[i][idTplCol - 1]) === String(idTemplate)) {
      sheet.getRange(i + 1, colIdx[campo]).setValue(valor);
      return;
    }
  }

  var novaLinha = new Array(headers.length).fill('');
  if (colIdx['IdResposta']) novaLinha[colIdx['IdResposta'] - 1] = generateShortId();
  novaLinha[idAgCol - 1]    = idAgendamento;
  novaLinha[idTplCol - 1]   = idTemplate;
  novaLinha[colIdx[campo] - 1] = valor;
  sheet.getRange(sheet.getLastRow() + 1, 1, 1, novaLinha.length).setValues([novaLinha]);
}

/**
 * action: salvarRespostasAgendamento — body { solicitanteId, idAgendamento, respostas:[{idTemplate,resposta}] }
 * Salva em lote as respostas de texto/número/sim-ou-não/lista do formulário
 * (fotos vão por uploadFotoResposta, que já salva sozinho).
 */
function actionSalvarRespostasAgendamento(body) {
  var idAgendamento = body.idAgendamento, respostas = body.respostas || [];
  if (!idAgendamento) return { ok: false, erro: 'idAgendamento é obrigatório.' };
  if (!isTecnicoOuAdmin(body.solicitanteId, idAgendamento)) {
    return { ok: false, erro: 'Sem permissão para essa ordem de serviço.' };
  }

  var tplSheet = getSheetSafe(CONFIG.ABA_TEMPLATES);
  var templates = tplSheet ? sheetToObjects(tplSheet) : [];
  var tplById = {};
  templates.forEach(function (t) { if (t.IdTemplate) tplById[t.IdTemplate] = t; });

  respostas.forEach(function (r) {
    if (!r.idTemplate) return;
    var tpl = tplById[r.idTemplate];
    var tipo = tpl ? String(tpl.TipoInput || '').trim() : 'Texto';
    var campo = (tipo === 'Number') ? 'RespostaQuantidade' : 'RespostaTexto';
    var valor = (r.resposta === undefined || r.resposta === null) ? '' : r.resposta;
    salvarOuAtualizarResposta(idAgendamento, r.idTemplate, campo, valor);
  });

  var novoStatus = atualizarStatusAutomatico(idAgendamento, templates);
  return { ok: true, novoStatus: novoStatus };
}

/**
 * Roda depois de qualquer resposta salva (texto/número/lista OU foto).
 * Regra: se ainda está "Agendado" e já tem pelo menos uma resposta -> "Em Andamento".
 * Se TODAS as perguntas obrigatórias do serviço atual já têm resposta -> "Concluído".
 * Nunca mexe em agendamento já Cancelado ou já Concluído (decisão manual é respeitada),
 * e nunca trava o formulário — só atualiza o status na planilha.
 */
/**
 * A coluna "Obrigatorio" da aba Templates é uma caixa de seleção (checkbox)
 * do Sheets — o Apps Script lê isso como BOOLEANO true/false, não como o
 * texto "VERDADEIRO"/"FALSO" que aparece na planilha. Essa função reconhece
 * os dois formatos.
 */
function isObrigatorioGS(valor) {
  if (valor === true) return true;
  if (valor === false || valor === undefined || valor === null || valor === '') return false;
  var s = String(valor).trim().toUpperCase();
  return s === 'VERDADEIRO' || s === 'TRUE' || s === 'SIM' || s === '1';
}

function atualizarStatusAutomatico(idAgendamento, templatesCache) {
  var found = findAgendamentoRowById(idAgendamento);
  if (!found) return null;

  var statusCol = found.colIdx['Status Agendamento'];
  var statusAtual = statusCol ? String(found.values[statusCol - 1] || '').trim() : '';
  var statusNorm = statusAtual.toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '');
  if (statusNorm === 'cancelado' || statusNorm === 'concluido') return null; // não mexe

  var idServico = found.values[found.colIdx['IdServico'] - 1];
  var templates = templatesCache;
  if (!templates) {
    var tplSheet = getSheetSafe(CONFIG.ABA_TEMPLATES);
    templates = tplSheet ? sheetToObjects(tplSheet) : [];
  }
  var obrigatorios = templates.filter(function (t) {
    return String(t.IdServico) === String(idServico) && isObrigatorioGS(t.Obrigatorio);
  });

  var todasRespostas = sheetToObjects(getOrCreateAgendamentosRespostasSheet())
    .filter(function (r) { return String(r.IdAgendamento) === String(idAgendamento); });
  var respostaPorTemplate = {};
  todasRespostas.forEach(function (r) { respostaPorTemplate[r.IdTemplate] = r; });

  function temValor(r) {
    return !!(r && (String(r.RespostaTexto || '').trim() || String(r.RespostaFoto || '').trim() || String(r.RespostaQuantidade || '').trim()));
  }

  var todosObrigatoriosRespondidos = obrigatorios.length > 0 && obrigatorios.every(function (t) {
    return temValor(respostaPorTemplate[t.IdTemplate]);
  });
  var temAlgumaResposta = todasRespostas.some(temValor);

  var novoStatus = null;
  if (todosObrigatoriosRespondidos) {
    novoStatus = 'Concluído';
  } else if (temAlgumaResposta && statusNorm === 'agendado') {
    novoStatus = 'Em Andamento';
  }

  if (novoStatus && novoStatus !== statusAtual) {
    found.sheet.getRange(found.rowIndex, statusCol).setValue(novoStatus);
    var statusNovoNorm = novoStatus.toLowerCase().normalize('NFD').replace(/[\u0300-\u036f]/g, '');
    if (statusNovoNorm === 'concluido' && found.colIdx['DataRelatorio']) {
      found.sheet.getRange(found.rowIndex, found.colIdx['DataRelatorio']).setValue(new Date());
    }
    return novoStatus;
  }
  return null;
}

/**
 * action: registrarPonto — body { solicitanteId, tipo, base64?, mimeType?, localizacao?, endereco? }
 * Usado pela tela de autoatendimento (Pontoeletronico.html) — o próprio
 * funcionário bate o ponto por ali. Tipo: "Entrada" | "Almoço" | "Retorno
 * Almoço" | "Saída". A selfie é opcional (algum funcionário pode não ter
 * câmera disponível), mas quando vem, sobe pro Drive igual às fotos de
 * agendamento.
 */
function actionRegistrarPonto(body) {
  var funcionario = body.solicitanteId, tipo = body.tipo;
  if (!funcionario || !tipo) return { ok: false, erro: 'Funcionário e tipo são obrigatórios.' };

  var vendedores = getSheetObjectsComCache(CONFIG.ABA_VENDEDORES, 'sh_vendedores');
  var pessoa = vendedores.filter(function (v) { return String(v.IdVendedor) === String(funcionario); })[0];
  if (!pessoa) return { ok: false, erro: 'Funcionário não encontrado.' };

  var fotoUrl = '';
  if (body.base64) {
    try {
      var bytes = Utilities.base64Decode(body.base64);
      var blob = Utilities.newBlob(bytes, body.mimeType || 'image/jpeg', (body.nomeArquivo || 'selfie') + '.jpg');
      var pasta = getOrCreatePastaFotosPonto();
      var arquivo = pasta.createFile(blob);
      arquivo.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
      fotoUrl = 'https://drive.google.com/thumbnail?id=' + arquivo.getId() + '&sz=w2000';
    } catch (e) {
      // se a foto falhar, ainda assim registra o ponto sem foto — melhor
      // bater o ponto sem selfie do que não bater ponto nenhum.
    }
  }

  var sheet = getOrCreateSheetComCabecalho(CONFIG.ABA_PONTO,
    ['IdPonto', 'Funcionario', 'Tipo', 'Data e Hora', 'Foto Selfie', 'Localizacao', 'Endereço']);
  var headers = getHeaderRow(sheet);
  var colIdx = {}; headers.forEach(function (h, i) { colIdx[h] = i + 1; });

  var novoId = body.idPonto || generateShortId(); // usa o ID que o cliente já gerou (otimista) — só gera aqui se não veio
  var lastCol = sheet.getLastColumn();
  var novaLinha = new Array(lastCol).fill('');
  var campos = {
    'IdPonto': novoId, 'Funcionario': funcionario, 'Tipo': tipo,
    'Data e Hora': new Date(), 'Foto Selfie': fotoUrl,
    'Localizacao': body.localizacao || '', 'Endereço': body.endereco || ''
  };
  Object.keys(colIdx).forEach(function (h) { if (campos[h] !== undefined) novaLinha[colIdx[h] - 1] = campos[h]; });
  sheet.getRange(sheet.getLastRow() + 1, 1, 1, novaLinha.length).setValues([novaLinha]);

  return { ok: true, idPonto: novoId, nome: pessoa.Nome || '', fotoUrl: fotoUrl };
}

function actionUploadFotoResposta(body) {

// ══════════════ ORDEM DE SERVIÇO EM PDF + ASSINATURA ELETRÔNICA (Autentique) ══════════════
//
// O "Relatório de Atendimento" (Ordem de Serviço) já é montado inteiro no
// navegador, em HTML — inclusive pra impressão direta (Ctrl+P). Pra poder
// mandar esse documento pra Autentique assinar, precisa de um PDF de
// verdade guardado em algum lugar (a Autentique não aceita HTML, e o
// navegador sozinho não gera PDF sem lib externa). Por isso esse fluxo é em
// duas etapas: 1) o navegador manda o MESMO HTML já montado pra cá, que só
// converte pra PDF e guarda no Drive; 2) com o fileId desse PDF em mãos, o
// navegador aciona o envio pra assinatura.

var AUTENTIQUE_GRAPHQL_URL_ = 'https://api.autentique.com.br/v2/graphql';

/**
 * Conta quantas páginas um PDF (Blob) tem, lendo os bytes brutos como texto
 * e procurando o "/Count N" do nó raiz da árvore de páginas — não existe
 * API nativa do Apps Script pra isso. Técnica documentada especificamente
 * pra PDF gerado pelo próprio Google (Docs/Sheets/HTML .getAs('application/
 * pdf')), que é justamente o nosso caso; não é garantido funcionar com PDF
 * de outra origem (ex.: um PDF já compactado com object streams, que esse
 * conversor do Google não usa).
 *
 * Necessário porque a Ordem de Serviço tem conteúdo de tamanho variável
 * (fotos/respostas do checklist) antes da página final de aceite/
 * assinatura, e o contrato também pode "vazar" pra uma página a mais ou a
 * menos dependendo do tamanho de endereço/cláusulas condicionais — sem
 * contar de verdade, não dá pra saber em que página a assinatura cai.
 * Fallback: se não achar o padrão (PDF vazio ou de 1 página só, às vezes
 * sem esse nó), assume 1 página.
 */
function contarPaginasPdf_(pdfBlob) {
  var texto = pdfBlob.getDataAsString('ISO-8859-1'); // preserva byte a byte, não reinterpreta como UTF-8
  var re = /\/Pages\s*\/Count\s+(\d+)/g, maior = 0, m;
  while ((m = re.exec(texto))) { var n = parseInt(m[1], 10); if (n > maior) maior = n; }
  var matchPages = texto.match(/\/Type\s*\/Page\b/g);
  var byObjects = matchPages ? matchPages.length : 0;
  return Math.max(maior, byObjects) || 1;
}

/**
 * Troca cada "https://drive.google.com/thumbnail?id=XXX&sz=..." que aparece
 * no HTML pelo conteúdo da própria imagem, embutido como "data:...;base64,"
 * — busca o arquivo direto pelo Drive (DriveApp.getFileById), sem depender
 * de rede nem do link público, que é onde a conversão HTML->PDF do Apps
 * Script falha de vez em quando. Cada fileId só é baixado uma vez, mesmo
 * que apareça repetido no HTML.
 */
function embutirFotosDriveComoBase64_(html) {
  var cache = {};
  return html.replace(/https:\/\/drive\.google\.com\/thumbnail\?id=([a-zA-Z0-9_-]+)(?:&amp;|&)sz=w\d+/g, function (match, fileId) {
    if (cache[fileId] !== undefined) return cache[fileId];
    try {
      var blob = DriveApp.getFileById(fileId).getBlob();
      var dataUri = 'data:' + blob.getContentType() + ';base64,' + Utilities.base64Encode(blob.getBytes());
      cache[fileId] = dataUri;
      return dataUri;
    } catch (err) {
      cache[fileId] = match; // não achou o arquivo — mantém o link original em vez de quebrar o PDF inteiro
      return match;
    }
  });
}

/**
 * action: gerarPdfOS — body { solicitanteId, idAgendamento, html }
 * "html" é o mesmo documento que a tela já monta pro botão "🖨 Gerar PDF"
 * (relatório de atendimento + a página de aceite/cláusula no final).
 * Converte pra PDF de verdade e salva na pasta "Solar Green - Ordens de
 * Serviço" do Drive — devolve fileId/url, que "enviarOSParaAssinatura" usa
 * em seguida. O link e a data ficam gravados na própria linha do
 * agendamento (colunas PDFrelatorio/PDFrelatorioFileId/DataRelatorio, pra
 * não precisar gerar de novo se alguém só quiser abrir o PDF depois).
 */
// A Ordem de Serviço (PDF + assinatura digital) só faz sentido depois do
// atendimento concluído — assinar "declarando que o serviço foi executado a
// contento" antes de ele acontecer não tem validade nenhuma. Réplica no
// servidor da mesma trava que já existe na tela, pra ninguém burlar chamando
// a API direto.
function agendamentoEstaConcluido_(found, statusAgendamento) {
  // statusAgendamento: mesmo motivo do isVendedorDoClienteOuAdmin_ acima —
  // agendamento pode só existir no Firestore, não na planilha. O navegador
  // já leu esse status certinho pra montar a própria tela.
  if (statusAgendamento !== undefined && statusAgendamento !== null) {
    return normalizarTextoGS(statusAgendamento) === 'concluido';
  }
  if (!found) return false;
  var col = found.colIdx['Status Agendamento'];
  if (!col) return false;
  return normalizarTextoGS(found.values[col - 1]) === 'concluido';
}

function actionGerarPdfOS(body) {
  var idAgendamento = body.idAgendamento, html = body.html;
  if (!idAgendamento || !html) return { ok: false, erro: 'idAgendamento e html são obrigatórios.' };
  if (!isVendedorDoClienteOuAdmin_(body.solicitanteId, idAgendamento, body.souAdmin, body.vendedorResponsavelCliente)) {
    return { ok: false, erro: 'Sem permissão para essa ordem de serviço.' };
  }
  if (!agendamentoEstaConcluido_(findAgendamentoRowById(idAgendamento), body.statusAgendamento)) {
    return { ok: false, erro: 'A Ordem de Serviço só pode ser gerada depois que o atendimento é marcado como Concluído.' };
  }

  try {
    var pasta = getOrCreatePastaOS_();
    var nomeArquivo = 'OS ' + idAgendamento + ' - ' + Utilities.formatDate(new Date(), Session.getScriptTimeZone() || 'GMT-3', 'dd-MM-yyyy HH-mm');

    // A conversão HTML->PDF do Apps Script não é confiável pra BUSCAR fotos
    // por link (https://drive.google.com/thumbnail?...) — é isso que
    // deixava algumas fotos "quebradas" no PDF mandado pra Autentique,
    // mesmo elas existindo normalmente no Drive. Solução: embutir cada foto
    // como base64 direto no HTML antes de converter, buscando o arquivo
    // pelo Drive (acesso direto, sem depender de rede/link) em vez de pedir
    // pro conversor buscar a URL sozinho.
    html = embutirFotosDriveComoBase64_(html);

    // Converter HTML->PDF direto pelo mimetype do blob é o único jeito de
    // gerar PDF sem lib externa no Apps Script — não dá pra usar o mesmo
    // caminho do Contrato (Google Docs + placeholders) porque a Ordem de
    // Serviço não vem de um modelo fixo, e sim montada dinamicamente com as
    // respostas/fotos de cada atendimento.
    var blobHtml = Utilities.newBlob(html, 'text/html', nomeArquivo + '.html');
    var pdfBlob = blobHtml.getAs('application/pdf').setName(nomeArquivo + '.pdf');
    var arquivo = pasta.createFile(pdfBlob);
    arquivo.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

    var found = findAgendamentoRowById(idAgendamento);
    if (found) {
      if (found.colIdx['PDFrelatorio'])       found.sheet.getRange(found.rowIndex, found.colIdx['PDFrelatorio']).setValue(arquivo.getUrl());
      if (found.colIdx['PDFrelatorioFileId']) found.sheet.getRange(found.rowIndex, found.colIdx['PDFrelatorioFileId']).setValue(arquivo.getId());
      if (found.colIdx['DataRelatorio'])      found.sheet.getRange(found.rowIndex, found.colIdx['DataRelatorio']).setValue(new Date());
    }

    return { ok: true, url: arquivo.getUrl(), fileId: arquivo.getId() };
  } catch (err) {
    return { ok: false, erro: 'Falha ao gerar o PDF da Ordem de Serviço: ' + err.message };
  }
}

/**
 * action: enviarOSParaAssinatura — body { solicitanteId, idAgendamento, fileId, clienteNome, clienteEmail, nomeDocumento }
 * Manda o PDF da Ordem de Serviço (já gerado no Drive por actionGerarPdfOS,
 * identificado por "fileId") pra Autentique via GraphQL, criando um
 * documento com um único signatário (o cliente). A própria Autentique
 * dispara o e-mail com o link de assinatura pro "clienteEmail" — como o
 * cliente já está no local com o técnico (celular/tablet), o normal é abrir
 * esse e-mail na hora e assinar ali mesmo. Este script só aciona a API e
 * devolve o link também, como cópia de backup (pra reenviar manualmente por
 * WhatsApp se precisar).
 *
 * O token da Autentique mora em Script Properties (AUTENTIQUE_API_TOKEN),
 * NUNCA no código-fonte — configure em ⚙️ Configurações do projeto >
 * Script Properties (veja o topo deste arquivo pro passo a passo).
 */
/**
 * Faz a parte em comum de mandar QUALQUER documento pra Autentique — um
 * único signatário, via GraphQL multipart (o "GraphQL multipart request
 * spec" que a Autentique exige pra upload de arquivo: 3 partes —
 * "operations" com a query+variáveis e o campo do arquivo como null, "map"
 * apontando qual parte do multipart preenche qual variável, e a parte com
 * o arquivo em si). Usada tanto pela Ordem de Serviço quanto pelo Contrato
 * — cada chamador cuida só de onde salvar o resultado (link/status/id).
 *
 * "posicoes" (opcional) — { cliente: [...], empresa: [...] }, no formato
 * de POSICOES_ASSINATURA_ acima. Quando informado, cada signatário ganha
 * marca visual (o "SignerInput.positions" da Autentique) na posição dada —
 * sem isso a assinatura sai "invisível" (só registro/certificado, sem
 * marca nenhuma no PDF), que era o comportamento de antes de 2026-09-03.
 * "z" pode vir como o texto 'ultima', resolvido aqui pra o número real da
 * última página do PDF (contarPaginasPdf_) — nunca cravar um número fixo,
 * porque os dois documentos têm tamanho variável antes da assinatura.
 *
 * Retorna { ok:false, erro } OU { ok:true, documento, link, sandbox }.
 */
// "config" no formato de POSICOES_ASSINATURA_[tipo][signatario] — ver o
// comentário lá em cima. Monta: se "assinaturaTodas" existir, 1 SIGNATURE
// completo por página de 1 até totalPaginas-1; senão, se "rubrica" existir,
// 1 INITIALS por página no mesmo intervalo; em documento de 1 página só,
// nenhum dos dois gera nada (não tem "página anterior"). Sempre termina com
// 1 SIGNATURE na última página (o "assinatura" obrigatório).
function resolverPosicoesAssinatura_(config, totalPaginas) {
  if (!config || !config.assinatura) return null;
  var out = [];
  if (config.assinaturaTodas) {
    for (var pg = 1; pg < totalPaginas; pg++) {
      out.push({ x: String(config.assinaturaTodas.x), y: String(config.assinaturaTodas.y), z: pg, element: 'SIGNATURE' });
    }
  } else if (config.rubrica) {
    for (var pg = 1; pg < totalPaginas; pg++) {
      out.push({ x: String(config.rubrica.x), y: String(config.rubrica.y), z: pg, element: 'INITIALS' });
    }
  }
  out.push({ x: String(config.assinatura.x), y: String(config.assinatura.y), z: totalPaginas, element: 'SIGNATURE' });
  return out;
}
// Identidade da Solar Green como SIGNATÁRIA de verdade na Autentique (não
// só texto no PDF) — precisa ser o mesmo e-mail dono do AUTENTIQUE_API_TOKEN
// configurado em Script Properties, senão o signDocument() (auto-assinatura)
// abaixo falha com "signature_not_found".
var EMPRESA_SIGNATARIO_EMAIL_ = 'solargreensuporte@gmail.com';
var EMPRESA_SIGNATARIO_NOME_  = 'Solar Green Suporte';

// Mesmo critério do EMAIL_ASSINATURA_RE em legado/proposta.html — mantenha os
// dois em sincronia. Vale pra Ordem de Serviço e pro Contrato (os dois passam
// por criarDocumentoNaAutentique_).
var EMAIL_SIGNATARIO_RE_ = /^[^\s@]+@[^\s@]+\.[A-Za-z]{2,}$/;

/**
 * Erro de validação da Autentique chega como message:"validation" — genérico,
 * sem dizer uma palavra sobre O QUE está errado (foi exatamente assim que o
 * Felipe recebeu só "Autentique: validation" ao enviar um contrato em
 * 2026-09-18, sem nenhuma pista pra agir). O motivo de verdade vem em
 * extensions.validation, um mapa campo -> [mensagens], que o código antigo
 * jogava fora. Aqui os dois são juntados, pra a tela mostrar algo como
 * "validation (signers.0.email: e-mail inválido)".
 */
function descreverErrosAutentique_(errors) {
  return (errors || []).map(function (e) {
    var msg = (e && e.message) ? String(e.message) : 'erro desconhecido';
    var validacao = e && e.extensions && e.extensions.validation;
    if (validacao && typeof validacao === 'object') {
      var detalhes = Object.keys(validacao).map(function (campo) {
        var lista = validacao[campo];
        return campo + ': ' + (Array.isArray(lista) ? lista.join(' / ') : String(lista));
      });
      if (detalhes.length) return msg + ' (' + detalhes.join(' | ') + ')';
    }
    return msg;
  }).join('; ');
}

// "partes" (opcional, 2026-10-22): [{ papel, nome, email }] para documento em
// que quem assina são as partes e a Solar Green fica de fora (vínculo com
// imóvel). Cada parte vira signatária com as posições de posicoes[papel] e
// recebe o próprio e-mail da Autentique; não há auto-assinatura. Sem "partes",
// tudo segue como antes (cliente + Solar Green assinando sozinha).
function criarDocumentoNaAutentique_(fileId, clienteNome, clienteEmail, nomeDocumento, posicoes, partes) {
  var props = PropertiesService.getScriptProperties();
  var token = props.getProperty('AUTENTIQUE_API_TOKEN');
  if (!token) return { ok: false, erro: 'Configure AUTENTIQUE_API_TOKEN em Project Settings > Script Properties.' };
  var sandbox = props.getProperty('AUTENTIQUE_SANDBOX') === 'true';

  if (partes) {
    partes = partes.map(function (p) {
      return { papel: String(p.papel || ''), nome: String(p.nome || '').trim(), email: String(p.email || '').trim() };
    });
    for (var i = 0; i < partes.length; i++) {
      if (!EMAIL_SIGNATARIO_RE_.test(partes[i].email)) {
        return { ok: false, erro: 'O e-mail do ' + partes[i].papel + ' ("' + partes[i].email + '") não é um endereço válido.' };
      }
      for (var j = 0; j < i; j++) {
        if (partes[j].email.toLowerCase() === partes[i].email.toLowerCase()) {
          return { ok: false, erro: 'Cada parte precisa de um e-mail diferente: a Autentique não aceita o mesmo signatário duas vezes no documento.' };
        }
      }
    }
  }

  // Signatário conferido aqui, no ponto por onde Ordem de Serviço e Contrato
  // passam os dois: a Autentique recusa e-mail malformado (e signatário
  // repetido) com um "validation" genérico, que não diz nem qual campo — o
  // erro fica muito mais claro conferindo antes. O trim também importa: com
  // espaço sobrando, o e-mail que a Autentique devolve não bate com o que
  // mandamos, e o link de assinatura voltava nulo.
  clienteEmail = String(clienteEmail || '').trim();
  clienteNome  = String(clienteNome || '').trim();
  if (!partes && !EMAIL_SIGNATARIO_RE_.test(clienteEmail)) {
    return { ok: false, erro: 'O e-mail do cliente ("' + clienteEmail + '") não é um endereço válido — corrija o cadastro do cliente antes de enviar pra assinatura.' };
  }
  if (!partes && clienteEmail.toLowerCase() === EMPRESA_SIGNATARIO_EMAIL_.toLowerCase()) {
    return { ok: false, erro: 'O e-mail cadastrado no cliente é o da própria Solar Green (' + EMPRESA_SIGNATARIO_EMAIL_ + '), e a Autentique não aceita o mesmo signatário duas vezes no documento. Cadastre o e-mail do cliente de verdade.' };
  }

  var blob;
  try {
    blob = DriveApp.getFileById(fileId).getBlob();
  } catch (err) {
    return { ok: false, erro: 'Não encontrei o PDF no Drive (fileId inválido).' };
  }

  var totalPaginas = contarPaginasPdf_(blob);

  // Dois signatários: o cliente (recebe e-mail da própria Autentique) e a
  // Solar Green (assinada automaticamente logo abaixo, via signDocument —
  // não precisa de ninguém clicar em nada do nosso lado). "positions" só
  // entra quando o chamador passou "posicoes" pra esse tipo de documento
  // (ver POSICOES_ASSINATURA_) — sem isso, cai no comportamento antigo
  // (assinatura sem marca visual, só o registro).
  var query = 'mutation CreateDocumentMutation($document: DocumentInput!, $signers: [SignerInput!]!, $file: Upload!) { ' +
    'createDocument(document: $document, signers: $signers, file: $file, sandbox: ' + (sandbox ? 'true' : 'false') + ') { ' +
    'id name signatures { public_id email link { short_link } } } }';
  var signerCliente = { email: clienteEmail, name: clienteNome || '', action: 'SIGN' };
  var signerEmpresa = { email: EMPRESA_SIGNATARIO_EMAIL_, name: EMPRESA_SIGNATARIO_NOME_, action: 'SIGN' };
  if (posicoes && !partes) {
    var posCliente = resolverPosicoesAssinatura_(posicoes.cliente, totalPaginas);
    var posEmpresa = resolverPosicoesAssinatura_(posicoes.empresa, totalPaginas);
    if (posCliente) signerCliente.positions = posCliente;
    if (posEmpresa) signerEmpresa.positions = posEmpresa;
  }
  var signersPartes = partes ? partes.map(function (p) {
    var signer = { email: p.email, name: p.nome, action: 'SIGN' };
    var pos = posicoes ? resolverPosicoesAssinatura_(posicoes[p.papel], totalPaginas) : null;
    if (pos) signer.positions = pos;
    return signer;
  }) : null;
  var variables = {
    document: { name: nomeDocumento || (clienteNome || 'Cliente') },
    signers: signersPartes || [signerCliente, signerEmpresa],
    file: null
  };
  var operations = JSON.stringify({ query: query, variables: variables });
  var map = JSON.stringify({ file: ['variables.file'] });

  var resposta = UrlFetchApp.fetch(AUTENTIQUE_GRAPHQL_URL_, {
    method: 'post',
    headers: { Authorization: 'Bearer ' + token },
    payload: { operations: operations, map: map, file: blob },
    muteHttpExceptions: true
  });

  var json;
  try {
    json = JSON.parse(resposta.getContentText());
  } catch (err) {
    return { ok: false, erro: 'Resposta inválida da Autentique: ' + resposta.getContentText() };
  }
  if (json.errors && json.errors.length) {
    return { ok: false, erro: 'Autentique: ' + descreverErrosAutentique_(json.errors) };
  }
  var documento = json.data && json.data.createDocument;
  if (!documento) return { ok: false, erro: 'Autentique não retornou o documento criado.' };
  if (partes) {
    // Sem signDocument: a Solar Green não é signatária deste documento. Os
    // links voltam só como cópia de reserva (a Autentique já mandou os e-mails).
    var links = partes.map(function (p) {
      var s = (documento.signatures || []).filter(function (x) { return String(x.email || '').toLowerCase() === p.email.toLowerCase(); })[0] || {};
      return s.link ? s.link.short_link : null;
    });
    return { ok: true, documento: documento, link: links.filter(Boolean).join(' '), links: links, sandbox: sandbox };
  }
  var assinaturaCliente = (documento.signatures || []).filter(function (s) { return s.email === clienteEmail; })[0] || {};
  var link = assinaturaCliente.link ? assinaturaCliente.link.short_link : null;

  // Assina automaticamente em nome da Solar Green (a conta dona do token)
  // — sem isso o documento ficaria "aguardando" as duas partes pra sempre,
  // já que ninguém entra na Autentique pra clicar em assinar manualmente.
  // Se falhar por qualquer motivo, não derruba o envio inteiro: o cliente
  // já recebeu o e-mail dele normalmente, só a nossa assinatura que fica
  // pendente (dá pra assinar depois direto pela Autentique, se precisar).
  try {
    var querySign = 'mutation { signDocument(id: ' + JSON.stringify(String(documento.id)) + ') }';
    UrlFetchApp.fetch(AUTENTIQUE_GRAPHQL_URL_, {
      method: 'post',
      headers: { Authorization: 'Bearer ' + token, 'Content-Type': 'application/json' },
      payload: JSON.stringify({ query: querySign }),
      muteHttpExceptions: true
    });
  } catch (err) { /* não é crítico — o documento já foi criado e enviado normalmente */ }

  return { ok: true, documento: documento, link: link, sandbox: sandbox };
}

function actionEnviarOSParaAssinatura(body) {
  var idAgendamento = body.idAgendamento, fileId = body.fileId,
      clienteNome = body.clienteNome, clienteEmail = body.clienteEmail;
  if (!fileId) return { ok: false, erro: 'Gere o PDF da Ordem de Serviço antes de enviar pra assinatura.' };
  if (!clienteEmail) return { ok: false, erro: 'O cliente precisa ter um e-mail cadastrado pra receber o link de assinatura.' };
  if (!isVendedorDoClienteOuAdmin_(body.solicitanteId, idAgendamento, body.souAdmin, body.vendedorResponsavelCliente)) {
    return { ok: false, erro: 'Sem permissão para essa ordem de serviço.' };
  }
  if (!agendamentoEstaConcluido_(findAgendamentoRowById(idAgendamento), body.statusAgendamento)) {
    return { ok: false, erro: 'A Ordem de Serviço só pode ser enviada pra assinatura depois que o atendimento é marcado como Concluído.' };
  }

  var resultado = criarDocumentoNaAutentique_(fileId, clienteNome, clienteEmail, body.nomeDocumento || ('Ordem de Serviço - ' + (clienteNome || 'Cliente')), POSICOES_ASSINATURA_.os);
  if (!resultado.ok) return resultado;
  var documento = resultado.documento, link = resultado.link;

  var found = findAgendamentoRowById(idAgendamento);
  if (found) {
    if (found.colIdx['LinkAssinaturaOS'])      found.sheet.getRange(found.rowIndex, found.colIdx['LinkAssinaturaOS']).setValue(link || '');
    if (found.colIdx['StatusAssinaturaOS'])    found.sheet.getRange(found.rowIndex, found.colIdx['StatusAssinaturaOS']).setValue('Enviado');
    if (found.colIdx['EnviadoAssinaturaOSEm']) found.sheet.getRange(found.rowIndex, found.colIdx['EnviadoAssinaturaOSEm']).setValue(new Date());
    if (found.colIdx['AutentiqueDocId'])       found.sheet.getRange(found.rowIndex, found.colIdx['AutentiqueDocId']).setValue(documento.id);
  }

  return { ok: true, autentiqueDocId: documento.id, link: link, sandbox: resultado.sandbox };
}

/**
 * action: verificarStatusOS — body { solicitanteId, idAgendamento }
 * Consulta a Autentique pelo status REAL do envio (não só "mandei e
 * pronto") — inclui se o e-mail chegou a ser entregue de verdade, porque
 * um envio pode "dar certo" do nosso lado e mesmo assim nunca chegar (caixa
 * de entrada cheia, e-mail bloqueado, etc — foi exatamente isso que
 * aconteceu com uma cliente e motivou essa checagem). Atualiza
 * StatusAssinaturaOS com o resultado, pra já ficar salvo pra próxima vez
 * que alguém abrir esse agendamento.
 */
function actionVerificarStatusOS(body) {
  var idAgendamento = body.idAgendamento;
  if (!idAgendamento) return { ok: false, erro: 'idAgendamento é obrigatório.' };
  if (!isVendedorDoClienteOuAdmin_(body.solicitanteId, idAgendamento, body.souAdmin, body.vendedorResponsavelCliente)) {
    return { ok: false, erro: 'Sem permissão para essa ordem de serviço.' };
  }

  var found = findAgendamentoRowById(idAgendamento);
  var autentiqueDocId = found && found.colIdx['AutentiqueDocId'] ? found.values[found.colIdx['AutentiqueDocId'] - 1] : '';
  if (!autentiqueDocId) return { ok: false, erro: 'Esta Ordem de Serviço ainda não foi enviada pra assinatura.' };

  // Precisa saber o e-mail do CLIENTE especificamente (não da Solar Green,
  // que agora também é signatária) pra achar a assinatura certa na resposta.
  var clienteEmailOS = '';
  if (found && found.colIdx['IdCliente']) {
    var idClienteOS = found.values[found.colIdx['IdCliente'] - 1];
    var clientesOS = sheetToObjects(getSheet(CONFIG.ABA_CLIENTES));
    var clienteObjOS = clientesOS.filter(function (c) { return String(c.IdCliente) === String(idClienteOS); })[0];
    clienteEmailOS = clienteObjOS ? String(clienteObjOS.Email || '') : '';
  }

  var props = PropertiesService.getScriptProperties();
  var token = props.getProperty('AUTENTIQUE_API_TOKEN');
  if (!token) return { ok: false, erro: 'Configure AUTENTIQUE_API_TOKEN em Project Settings > Script Properties.' };

  // O id vai direto no corpo da query (como no exemplo oficial da Autentique,
  // "document(id: \"DOCUMENT ID\")"), em vez de variável tipada — assim não
  // depende de acertar o nome exato do scalar type (UUID/ID/String) no
  // schema deles. JSON.stringify escapa o valor com segurança pra dentro da
  // string da query.
  var query = 'query { document(id: ' + JSON.stringify(String(autentiqueDocId)) + ') { signatures { ' +
    'email link { short_link } email_events { sent opened delivered refused reason } ' +
    'viewed { created_at } signed { created_at } rejected { created_at } } } }';
  var resposta = UrlFetchApp.fetch(AUTENTIQUE_GRAPHQL_URL_, {
    method: 'post',
    headers: { Authorization: 'Bearer ' + token, 'Content-Type': 'application/json' },
    payload: JSON.stringify({ query: query }),
    muteHttpExceptions: true
  });

  var json;
  try {
    json = JSON.parse(resposta.getContentText());
  } catch (err) {
    return { ok: false, erro: 'Resposta inválida da Autentique: ' + resposta.getContentText() };
  }
  if (json.errors && json.errors.length) {
    return { ok: false, erro: 'Autentique: ' + descreverErrosAutentique_(json.errors) };
  }
  var documento = json.data && json.data.document;
  if (!documento) return { ok: false, erro: 'Documento não encontrado na Autentique.' };
  var assinatura = (documento.signatures || []).filter(function (s) { return s.email === clienteEmailOS; })[0]
    || (documento.signatures || [])[0] || {};
  var eventos = assinatura.email_events || {};

  var statusTexto, refused = !!(eventos.refused || eventos.reason);
  if (assinatura.signed) statusTexto = 'Assinado';
  else if (assinatura.rejected) statusTexto = 'Recusado pelo cliente';
  else if (refused) statusTexto = 'E-mail não entregue' + (eventos.reason ? (': ' + eventos.reason) : '');
  else if (assinatura.viewed) statusTexto = 'Visualizado, aguardando assinatura';
  else if (eventos.delivered) statusTexto = 'E-mail entregue, aguardando abertura';
  else statusTexto = 'Enviado, aguardando confirmação de entrega';

  if (found && found.colIdx['StatusAssinaturaOS']) {
    found.sheet.getRange(found.rowIndex, found.colIdx['StatusAssinaturaOS']).setValue(statusTexto);
  }

  return {
    ok: true, status: statusTexto, assinado: !!assinatura.signed, refused: refused,
    reason: eventos.reason || null, link: assinatura.link ? assinatura.link.short_link : null
  };
}

function getOrCreateContratosAssinadosSheet_() {

    sheet.getRange(1, 1, 1, CONTRATOS_ASSINADOS_HEADERS.length).setValues([CONTRATOS_ASSINADOS_HEADERS]);
    sheet.setFrozenRows(1);
  }
  return sheet;
}

  return DriveApp.createFolder(nomePasta);
}

var TIPOS_CONTRATO_ASSINAVEIS_ = { monitoramento: true, manutencao: true, recibo: true, procuracao: true, homologacao: true, vinculo_imovel: true };

// Tipos em que quem assina são as partes do contrato, e não cliente + Solar
// Green (2026-10-22, vínculo com imóvel). A tela manda body.signatarios com
// um item por papel daqui.
var PARTES_SIGNATARIAS_ = { vinculo_imovel: ['locador', 'locatario'] };

/**
 * action: enviarContratoParaAssinatura — body { idCliente, clienteNome,
 *   clienteEmail, tipoContrato, html, nomeDocumento }
 * "html" é o MESMO documento que a tela já monta pro botão "Imprimir/PDF"
 * do Contrato (compose* de cada tipo). Só os tipos em
 * TIPOS_CONTRATO_ASSINAVEIS_ podem ser mandados pra assinatura — liberado
 * pra Recibo e Procuração em 2026-08-27 (pedido do Felipe; antes só
 * monitoramento/manutenção, sem motivo técnico pra excluir os outros dois,
 * o fluxo já é genérico).
 */
function actionEnviarContratoParaAssinatura(body) {
  var tipoContrato = body.tipoContrato, html = body.html,
      clienteNome = body.clienteNome, clienteEmail = body.clienteEmail;
  if (!html) return { ok: false, erro: 'Faltam os dados do contrato.' };
  if (!TIPOS_CONTRATO_ASSINAVEIS_[tipoContrato]) {
    return { ok: false, erro: 'Esse tipo de documento não pode ser enviado pra assinatura digital.' };
  }
  var partes = null, papeis = PARTES_SIGNATARIAS_[tipoContrato];
  if (papeis) {
    partes = papeis.map(function (papel) {
      return (body.signatarios || []).filter(function (s) { return s && s.papel === papel; })[0];
    });
    if (partes.some(function (p) { return !p; })) {
      return { ok: false, erro: 'Faltam os dados de quem assina (' + papeis.join(' e ') + ').' };
    }
  } else if (!clienteEmail) return { ok: false, erro: 'O cliente precisa ter um e-mail cadastrado pra receber o link de assinatura.' };

  try {
    var pasta = getOrCreatePastaContratosProposta_();
    var nomeArquivo = 'Contrato - ' + (clienteNome || 'Cliente') + ' - ' + Utilities.formatDate(new Date(), Session.getScriptTimeZone() || 'GMT-3', 'dd-MM-yyyy HH-mm');

    html = embutirFotosDriveComoBase64_(html);
    var blobHtml = Utilities.newBlob(html, 'text/html', nomeArquivo + '.html');
    var pdfBlob = blobHtml.getAs('application/pdf').setName(nomeArquivo + '.pdf');
    var arquivo = pasta.createFile(pdfBlob);
    arquivo.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

    // Recibo/Procuração ainda não têm posição configurada (ver comentário
    // de POSICOES_ASSINATURA_) — undefined aqui faz criarDocumentoNaAutentique_
    // cair no comportamento antigo pra esses dois tipos (sem marca visual),
    // em vez de dar erro.
    var resultado = criarDocumentoNaAutentique_(arquivo.getId(), clienteNome, clienteEmail, body.nomeDocumento || ('Contrato - ' + (clienteNome || 'Cliente')), POSICOES_ASSINATURA_[tipoContrato], partes);
    if (!resultado.ok) return resultado;
    var documento = resultado.documento, link = resultado.link;

    var sheet = getOrCreateContratosAssinadosSheet_();
    sheet.appendRow([
      generateShortId(), body.idCliente || '', clienteNome || '', clienteEmail,
      tipoContrato, arquivo.getUrl(), arquivo.getId(), link || '', 'Enviado', documento.id, new Date()
    ]);

    return { ok: true, url: arquivo.getUrl(), fileId: arquivo.getId(), autentiqueDocId: documento.id, link: link };
  } catch (err) {
    return { ok: false, erro: 'Falha ao gerar/enviar o contrato pra assinatura: ' + err.message };
  }
}

/**
 * Rode MANUALMENTE uma vez pelo editor (▶ Executar). O Apps Script só pede a
 * permissão de uma operação quando ela é realmente executada — por isso essa
 * função não só LÊ o Drive, mas também CRIA uma pasta/arquivo de teste e muda
 * o compartilhamento dele, pra garantir que a permissão de ESCRITA (a que
 * faltava) também seja pedida e concedida de uma vez. A pasta de teste é
 * apagada (lixeira) no final — não fica lixo no seu Drive.
 *
 * A chamada UrlFetchApp aqui no final é só pra forçar o Google a listar o
 * escopo "script.external_request" na tela de autorização — sem isso, quem
 * já tinha autorizado antes de existir a assinatura eletrônica (que é quem

 * permissão de novo, e a chamada falha com "Você não tem permissão para
 * chamar UrlFetchApp.fetch" mesmo depois de reimplantado. Rodar esta função
 * de novo (▶) e aceitar a tela de permissão resolve — não precisa
 * reautorizar toda vez, só quando um escopo novo é adicionado.
 */
function autorizarPermissoes() {
  DriveApp.getRootFolder();

  pastaTeste.setTrashed(true);

  MailApp.getRemainingDailyQuota();
  UrlFetchApp.fetch('https://www.google.com', { muteHttpExceptions: true });
  return 'Permissões OK (leitura, escrita e compartilhamento no Drive + Mail + acesso externo).';
}

function colIdxDaAba(sheet) {
  var headers = sheet.getRange(1, 1, 1, sheet.getLastColumn()).getValues()[0].map(function (h) { return String(h).trim(); });
  var colIdx = {};
  headers.forEach(function (h, i) { if (h) colIdx[h] = i + 1; });
  return colIdx;
}

/**
 * action: getCatalogoProdutos — sem parâmetros além da chave.
 * Inversores já vêm com mpptsConfig parseado em array (não a string JSON crua).
 */
function getCatalogoProdutos() {
  var modulos = sheetToObjects(getOrCreateProdutosModulosSheet());
  var inversores = sheetToObjects(getOrCreateProdutosInversoresSheet()).map(function (inv) {
    var out = {};
    Object.keys(inv).forEach(function (k) { if (k !== 'MpptsConfigJSON') out[k] = inv[k]; });
    try {
      out.mpptsConfig = inv.MpptsConfigJSON ? JSON.parse(inv.MpptsConfigJSON) : [];
    } catch (e) {
      out.mpptsConfig = [];
    }
    return out;
  });
  return { ok: true, modulos: modulos, inversores: inversores };
}

/**
 * action: salvarModulo — body { idModulo?, marca, modelo, alturaM, larguraM,
 *   potenciaW, vocV, iscA, pesoKg }
 */
function actionSalvarModulo(body) {
  var modelo = String(body.modelo || '').trim();
  if (!modelo) return { ok: false, erro: 'Modelo do módulo é obrigatório.' };

  var sheet = getOrCreateProdutosModulosSheet();
  var colIdx = colIdxDaAba(sheet);

  var rowIndex = -1;
  if (body.idModulo && colIdx['IdModulo']) {
    var values = sheet.getDataRange().getValues();
    for (var i = 1; i < values.length; i++) {
      if (String(values[i][colIdx['IdModulo'] - 1]) === String(body.idModulo)) { rowIndex = i + 1; break; }
    }
  }

  var agora = new Date();
  var campos = {
    'Marca': body.marca || '',
    'Modelo': modelo,
    'AlturaM': paraNumero(body.alturaM),
    'LarguraM': paraNumero(body.larguraM),
    'PotenciaW': paraNumero(body.potenciaW),
    'VocV': paraNumero(body.vocV),
    'IscA': paraNumero(body.iscA),
    'PesoKg': paraNumero(body.pesoKg),
    'DataAtualizacao': agora
  };

  var novoId = body.idModulo;
  if (rowIndex === -1) {
    novoId = novoId || generateShortId();
    campos['DataCriacao'] = agora;
    var lastCol = sheet.getLastColumn();
    var novaLinha = new Array(lastCol).fill('');
    Object.keys(colIdx).forEach(function (h) {
      if (h === 'IdModulo') novaLinha[colIdx[h] - 1] = novoId;
      else if (campos[h] !== undefined) novaLinha[colIdx[h] - 1] = campos[h];
    });
    sheet.getRange(sheet.getLastRow() + 1, 1, 1, novaLinha.length).setValues([novaLinha]);
  } else {
    Object.keys(campos).forEach(function (h) { if (colIdx[h]) sheet.getRange(rowIndex, colIdx[h]).setValue(campos[h]); });
  }
  return { ok: true, idModulo: novoId };
}

    return { ok: false, erro: 'Acesso restrito a administradores.' };

  var colIdx = colIdxDaAba(sheet);

//  NOTIFICAÇÃO DE LEAD DO SITE (2026-10-08)
// =====================================================================
// O site grava o contato em leads_site e chama "avisarLeadSite". Aqui o
// vendedor da vez é sorteado (mesma regra e mesmo documento de rodízio que o
// ERP usa em js/leads-site-motor.js) e sai o push pro celular/computador do
// vendedor e da gestão, mesmo com o ERP fechado. Quem cria o cliente e o lead
// no Funil continua sendo o ERP, com o vendedor que foi sorteado aqui.
//
// O ERP também chama esta ação depois de promover, marcar suspeito ou dar
// erro. Cada aviso sai uma vez só (campos avisoLeadEm e avisoAtencaoEm).
//
// Precisa de uma Script Property:
//   FIREBASE_SERVICE_ACCOUNT = o conteúdo inteiro do .json da conta de serviço
//                              do Firebase (solargreen-21313-firebase-adminsdk-...)
// Depois de configurar, rode "conferirNotificacoes" pelo editor pra ver se a
// conta de serviço entra no Firestore e quantos aparelhos estão cadastrados.

// Regra de vendedor no rodízio: igual a isVendedorAtivo em js/vendas.js.
function ehVendedorAtivoFs_(v) {
  return String(v.Tipo || '').trim() === 'Vendedor' && String(v.Status || '').trim() === 'Ativo';
}
function ehGestaoAtivaFs_(v) {
  return ADMIN_ROLES.indexOf(String(v.Tipo || '').trim().toLowerCase()) !== -1 && String(v.Status || '').trim() === 'Ativo';
}

function actionAvisarLeadSite(body) {
  var id = String(body.id || '');
  if (!/^[A-Za-z0-9_-]{1,64}$/.test(id)) return { ok: false, erro: 'id inválido' };
  var cfg = fbConfig_();
  if (!cfg) return { ok: false, erro: 'FIREBASE_SERVICE_ACCOUNT não configurada' };
  // Teto de chamadas por minuto: o site manda uma por contato, o ERP uma por
  // promoção. Muito acima disso é alguém chamando em loop pra gastar a cota.
  var cache = CacheService.getScriptCache();
  var minuto = 'sg_aviso_' + Math.floor(Date.now() / 60000);
  var feitas = Number(cache.get(minuto) || 0);
  if (feitas >= 40) return { ok: false, erro: 'muitas chamadas, tente em um minuto' };
  cache.put(minuto, String(feitas + 1), 120);
  // Um aviso por vez: o rodízio depende de um sorteio terminar antes do próximo.
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(25000)) return { ok: false, erro: 'servidor ocupado' };
  try {
    for (var tentativa = 0; tentativa < 3; tentativa++) {
      var r = avisarLeadSiteUmaVez_(cfg, id);
      if (r !== 'repetir') return r;
    }
    return { ok: false, erro: 'o contato mudou durante o sorteio três vezes seguidas' };
  } finally {
    lock.releaseLock();
  }
}

function avisarLeadSiteUmaVez_(cfg, id) {
  var doc = fsGet_(cfg, 'leads_site/' + id);
  if (!doc) return { ok: false, erro: 'contato não encontrado' };
  var l = doc.dados;
  var agora = Date.now();
  // Nada a avisar: sai antes de ler a lista de vendedores. A ação é pública,
  // e cada leitura conta na cota diária do Firestore.
  var atencao = (l.suspeito === true && (l.status === 'novo' || l.status === 'suspeito')) || l.status === 'erro';
  var ehLead = l.status === 'novo' || l.status === 'processando' || l.status === 'promovido';
  if (atencao ? l.avisoAtencaoEm : (!ehLead || l.avisoLeadEm)) return { ok: true, acao: (atencao || ehLead) ? 'já avisado' : 'nada a avisar' };
  var vendedores = vendedoresFs_(cfg);
  var nomes = {}, gestao = [];
  vendedores.forEach(function (v) {
    nomes[String(v.IdVendedor)] = v.Nome || '';
    if (ehGestaoAtivaFs_(v)) gestao.push(String(v.IdVendedor));
  });
  var resumo = [l.nome || 'Sem nome', l.servico, l.cidade || l.calcCidade].filter(Boolean).join(' · ');

  // Suspeito ou erro: só a gestão resolve, aviso vai só pra ela.
  var suspeito = l.suspeito === true && (l.status === 'novo' || l.status === 'suspeito');
  if (suspeito || l.status === 'erro') {
    if (l.avisoAtencaoEm) return { ok: true, acao: 'já avisado' };
    var tipo = l.status === 'erro' ? 'erro' : 'suspeito';
    enviarPush_(cfg, gestao, {
      titulo: tipo === 'erro' ? 'Contato do site não virou lead' : 'Contato suspeito no site',
      corpo: resumo + ' · confira em Leads do site',
      url: 'index.html?tela=leadssite&notif=' + id + ':' + tipo,
      tag: 'lead-' + id, notifId: id + ':' + tipo
    });
    fsCommit_(cfg, [fsUpdate_(cfg, 'leads_site/' + id, { avisoAtencaoEm: agora })]);
    return { ok: true, acao: tipo };
  }

  // Sorteio do vendedor da vez.
  if (l.status === 'novo' && !l.idVendedorSorteado) {
    var ativos = vendedores.filter(ehVendedorAtivoFs_).sort(function (a, b) {
      return String(a.IdVendedor).localeCompare(String(b.IdVendedor));
    });
    // Sem vendedor ativo, o ERP atribui a quem estiver com ele aberto e chama
    // de novo depois de promover; aí o aviso sai pelo caminho de baixo.
    if (!ativos.length) return { ok: true, acao: 'sem vendedor ativo' };
    var rod = fsGet_(cfg, 'config/rodizio_site');
    var ultimo = rod ? String(rod.dados.ultimoIdVendedor || '') : '';
    var idx = -1;
    ativos.forEach(function (v, i) { if (String(v.IdVendedor) === ultimo) idx = i; });
    var prox = String(ativos[(idx + 1) % ativos.length].IdVendedor);
    var idOp = l.idOportunidade || Utilities.getUuid().replace(/-/g, '').substring(0, 8);
    // Se o ERP mexeu no contato ou no rodízio entre a leitura e aqui, o commit
    // é recusado e a função roda de novo com o dado novo.
    var ok = fsCommit_(cfg, [
      fsUpdate_(cfg, 'leads_site/' + id, { idVendedorSorteado: prox, sorteadoEm: agora, idOportunidade: idOp }, { updateTime: doc.updateTime }),
      fsUpdate_(cfg, 'config/rodizio_site', { ultimoIdVendedor: prox, atualizadoEm: agora }, rod ? { updateTime: rod.updateTime } : { exists: false })
    ]);
    if (!ok) return 'repetir';
    l.idVendedorSorteado = prox;
    l.idOportunidade = idOp;
  }

  // Aviso do lead: vendedor da vez e gestão.
  var vendedor = l.status === 'promovido' ? String(l.idVendedor || '') : String(l.idVendedorSorteado || '');
  if (!vendedor || (l.status !== 'promovido' && l.status !== 'novo' && l.status !== 'processando')) return { ok: true, acao: 'nada a avisar' };
  if (l.avisoLeadEm) return { ok: true, acao: 'já avisado' };
  var url = 'index.html?lead=' + (l.idOportunidade || '') + '&notif=' + id + ':lead';
  enviarPush_(cfg, [vendedor], { titulo: 'Novo lead do site pra você', corpo: resumo, url: url, tag: 'lead-' + id, notifId: id + ':lead' });
  enviarPush_(cfg, gestao.filter(function (g) { return g !== vendedor; }), {
    titulo: 'Novo lead do site', corpo: resumo + ' · para ' + (nomes[vendedor] || 'vendedor da vez'),
    url: url, tag: 'lead-' + id, notifId: id + ':lead'
  });
  fsCommit_(cfg, [fsUpdate_(cfg, 'leads_site/' + id, { avisoLeadEm: agora })]);
  return { ok: true, acao: 'lead', vendedor: vendedor };
}

// Lista de vendedores guardada por 1 minuto: vários contatos seguidos não
// releem a coleção a cada um.
function vendedoresFs_(cfg) {
  var cache = CacheService.getScriptCache();
  var guardado = cache.get('sg_vendedores_fs');
  if (guardado) return JSON.parse(guardado);
  var lista = fsListar_(cfg, 'vendedores').map(function (v) {
    return { IdVendedor: v.IdVendedor, Nome: v.Nome, Tipo: v.Tipo, Status: v.Status };
  });
  cache.put('sg_vendedores_fs', JSON.stringify(lista), 60);
  return lista;
}

// Manda pelo Firebase Cloud Messaging pra todos os aparelhos das pessoas da
// lista. Aparelho que o Firebase diz que não existe mais sai de push_tokens.
function enviarPush_(cfg, idsVendedores, msg) {
  if (!idsVendedores.length) return 0;
  var aparelhos = fsListar_(cfg, 'push_tokens').filter(function (t) {
    return idsVendedores.indexOf(String(t.idVendedor)) !== -1 && t.token;
  });
  if (!aparelhos.length) return 0;
  var tok = fbAccessToken_(cfg);
  var pedidos = aparelhos.map(function (t) {
    return {
      url: 'https://fcm.googleapis.com/v1/projects/' + cfg.projeto + '/messages:send',
      method: 'post', contentType: 'application/json', muteHttpExceptions: true,
      headers: { Authorization: 'Bearer ' + tok },
      payload: JSON.stringify({ message: {
        token: t.token,
        data: { titulo: msg.titulo, corpo: msg.corpo, url: msg.url, tag: msg.tag, notifId: msg.notifId },
        webpush: { headers: { Urgency: 'high', TTL: '86400' } }
      } })
    };
  });
  var enviados = 0;
  UrlFetchApp.fetchAll(pedidos).forEach(function (resp, i) {
    var cod = resp.getResponseCode();
    if (cod === 200) { enviados++; return; }
    var texto = resp.getContentText();
    if (cod === 404 || texto.indexOf('UNREGISTERED') !== -1) {
      fsApagar_(cfg, 'push_tokens/' + aparelhos[i]._id);
    } else {
      console.error('Push recusado (' + cod + '): ' + texto.slice(0, 300));
    }
  });
  return enviados;
}

// ── Firestore por REST, com a conta de serviço ─────────────────────────

function fbConfig_() {
  var bruto = PropertiesService.getScriptProperties().getProperty('FIREBASE_SERVICE_ACCOUNT');
  if (!bruto) return null;
  var sa = JSON.parse(bruto);
  return {
    projeto: sa.project_id, email: sa.client_email, chave: sa.private_key,
    base: 'https://firestore.googleapis.com/v1/projects/' + sa.project_id + '/databases/(default)/documents'
  };
}

function fbAccessToken_(cfg) {
  var cache = CacheService.getScriptCache();
  var guardado = cache.get('sg_fb_token');
  if (guardado) return guardado;
  var agora = Math.floor(Date.now() / 1000);
  function b64(x) { return Utilities.base64EncodeWebSafe(x).replace(/=+$/, ''); }
  var cab = b64(JSON.stringify({ alg: 'RS256', typ: 'JWT' }));
  var corpo = b64(JSON.stringify({
    iss: cfg.email, aud: 'https://oauth2.googleapis.com/token', iat: agora, exp: agora + 3600,
    scope: 'https://www.googleapis.com/auth/datastore https://www.googleapis.com/auth/firebase.messaging'
  }));
  var assinatura = b64(Utilities.computeRsaSha256Signature(cab + '.' + corpo, cfg.chave));
  var resp = UrlFetchApp.fetch('https://oauth2.googleapis.com/token', {
    method: 'post', muteHttpExceptions: true,
    payload: { grant_type: 'urn:ietf:params:oauth:grant-type:jwt-bearer', assertion: cab + '.' + corpo + '.' + assinatura }
  });
  if (resp.getResponseCode() !== 200) throw new Error('Google recusou a conta de serviço: ' + resp.getContentText().slice(0, 200));
  var token = JSON.parse(resp.getContentText()).access_token;
  cache.put('sg_fb_token', token, 3000);
  return token;
}

function fsPedir_(cfg, metodo, url, corpo) {
  var opcoes = { method: metodo, muteHttpExceptions: true, headers: { Authorization: 'Bearer ' + fbAccessToken_(cfg) } };
  if (corpo) { opcoes.contentType = 'application/json'; opcoes.payload = JSON.stringify(corpo); }
  return UrlFetchApp.fetch(url, opcoes);
}

function fsLerValor_(v) {
  if (v.hasOwnProperty('stringValue')) return v.stringValue;
  if (v.hasOwnProperty('integerValue')) return Number(v.integerValue);
  if (v.hasOwnProperty('doubleValue')) return v.doubleValue;
  if (v.hasOwnProperty('booleanValue')) return v.booleanValue;
  if (v.hasOwnProperty('timestampValue')) return Date.parse(v.timestampValue);
  if (v.hasOwnProperty('mapValue')) return fsLerCampos_(v.mapValue.fields || {});
  if (v.hasOwnProperty('arrayValue')) return (v.arrayValue.values || []).map(fsLerValor_);
  return null;
}
function fsLerCampos_(campos) {
  var out = {};
  Object.keys(campos).forEach(function (k) { out[k] = fsLerValor_(campos[k]); });
  return out;
}
function fsValor_(v) {
  if (typeof v === 'boolean') return { booleanValue: v };
  if (typeof v === 'number') return Number.isInteger(v) ? { integerValue: String(v) } : { doubleValue: v };
  return { stringValue: String(v) };
}

function fsGet_(cfg, caminho) {
  var resp = fsPedir_(cfg, 'get', cfg.base + '/' + caminho);
  if (resp.getResponseCode() === 404) return null;
  if (resp.getResponseCode() !== 200) throw new Error('Firestore recusou a leitura de ' + caminho + ': ' + resp.getContentText().slice(0, 200));
  var d = JSON.parse(resp.getContentText());
  return { dados: fsLerCampos_(d.fields || {}), updateTime: d.updateTime };
}

function fsListar_(cfg, colecao) {
  var lista = [], pagina = '';
  do {
    var resp = fsPedir_(cfg, 'get', cfg.base + '/' + colecao + '?pageSize=300' + (pagina ? '&pageToken=' + encodeURIComponent(pagina) : ''));
    if (resp.getResponseCode() !== 200) throw new Error('Firestore recusou a lista de ' + colecao + ': ' + resp.getContentText().slice(0, 200));
    var d = JSON.parse(resp.getContentText());
    (d.documents || []).forEach(function (doc) {
      var item = fsLerCampos_(doc.fields || {});
      item._id = doc.name.split('/').pop();
      lista.push(item);
    });
    pagina = d.nextPageToken || '';
  } while (pagina);
  return lista;
}

// Escrita que só mexe nos campos informados. `condicao` é a pré-condição do
// Firestore ({updateTime} ou {exists:false}); sem ela, grava de qualquer jeito.
function fsUpdate_(cfg, caminho, campos, condicao) {
  var fields = {};
  Object.keys(campos).forEach(function (k) { fields[k] = fsValor_(campos[k]); });
  var w = {
    update: { name: cfg.base.replace('https://firestore.googleapis.com/v1/', '') + '/' + caminho, fields: fields },
    updateMask: { fieldPaths: Object.keys(campos) }
  };
  if (condicao) w.currentDocument = condicao;
  return w;
}

// Devolve false quando uma pré-condição falhou (alguém mexeu antes).
function fsCommit_(cfg, writes) {
  var resp = fsPedir_(cfg, 'post', cfg.base + ':commit', { writes: writes });
  var cod = resp.getResponseCode();
  if (cod === 200) return true;
  var texto = resp.getContentText();
  if (texto.indexOf('FAILED_PRECONDITION') !== -1 || texto.indexOf('ABORTED') !== -1 || cod === 409) return false;
  throw new Error('Firestore recusou a gravação: ' + texto.slice(0, 300));
}

function fsApagar_(cfg, caminho) {
  fsPedir_(cfg, 'delete', cfg.base + '/' + caminho);
}

function disparoEnviarLote() {
  var props = disparoProps_();
  // O prazo é conferido por disparo: a campanha pode ter o dela (campanhas/{id}.prazo).
  if (!props.ativo || !props.token || !props.phoneId || !props.segredo) return;

  // acionador roda a cada 15 minutos; depois de DISPARO_HORA_FIM só atualiza a
  // planilha, uma vez por hora, pra tarde mostrar quem interagiu.
  var inicio = Date.now();
  var agoraTxt = Utilities.formatDate(new Date(), 'America/Belem', 'u H m yyyy-MM-dd').split(' ');
  var diaSemana = Number(agoraTxt[0]), hora = Number(agoraTxt[1]), minuto = Number(agoraTxt[2]), hoje = agoraTxt[3];
  if (diaSemana > 5 || hora < 8 || hora > 17) return;

    if (minuto < 15) try { disparoAtualizarPlanilha(); } catch (e) { console.error('Planilha do disparo: ' + e.message); }
    return;
  }
  var lock = LockService.getScriptLock();
  if (!lock.tryLock(1000)) return;
  try {
    var cfg = fbConfig_();
    if (!cfg) return;

    // e toda mensagem voltou com erro): espera 1 hora antes de tentar de novo,
    // em vez de gastar a fila inteira em erro.

    var cont = fsGet_(cfg, 'config/disparo_contagem');
    var porVendedor = cont && cont.dados.data === hoje ? (cont.dados.porVendedor || {}) : {};
    // A cota do dia se espalha pelas rodadas de 15 minutos que faltam até DISPARO_HORA_FIM.
    var rodadasRestantes = Math.max(1, Math.ceil((props.horaFim * 60 - (hora * 60 + minuto)) / 15));
    var livres = props.limite24h - disparoEnviados24h_(cfg);
    // Campanhas lidas uma vez por rodada (coleção pequena). Só campanha "ativa" envia.
    var campanhas = {};

    disparoAgendarPassos_(cfg, campanhas);

    Object.keys(campanhas).forEach(function (id) {

    });

      // Contato tirado da campanha pela tela (removidoEm) não recebe.

      var restante = props.cota - feitos;
      if (restante <= 0) return;
      var nestaRodada = Math.ceil(restante / rodadasRestantes);
      if ((enviadosAgora[v] || 0) >= nestaRodada) return;
      if (fsGet_(cfg, 'disparo_optout/' + disparoFim8_(d.telLead))) {
        fsCommit_(cfg, [fsUpdate_(cfg, 'disparos/' + d._id, { status: 'saiu' })]);
        return;

        var docModelo = fsGet_(cfg, 'modelos_disparo/' + d.modelo);
        modelos[d.modelo] = docModelo ? docModelo.dados : null;
      }
      var r = disparoMandarModelo_(props, d, prazo, modelos[d.modelo]);
      if (r.ok) {
        fsCommit_(cfg, [fsUpdate_(cfg, 'disparos/' + d._id, { status: 'enviado', wamid: r.wamid, enviadoEm: Date.now() })]);
        enviadosAgora[v] = (enviadosAgora[v] || 0) + 1;
        livres--;
        porVendedor[v] = feitos + 1;
      } else if (r.codigo === 131049) {
        // A Meta segurou por excesso de propaganda pra essa pessoa. Não cobra.
        fsCommit_(cfg, [fsUpdate_(cfg, 'disparos/' + d._id, { naoAntesDe: Date.now() + 26 * 3600 * 1000, erro: r.erro })]);
      } else if (r.codigo === DISPARO_ERRO_PAGAMENTO) {
        // O contato não tem culpa: fica na fila e a rodada para.
        disparoPagamentoFalhou_(cfg);
        livres = 0;
      } else {
        fsCommit_(cfg, [fsUpdate_(cfg, 'disparos/' + d._id, { status: 'erro', erro: r.erro, erroCodigo: r.codigo || 0 })]);
      }
    });
    var w = fsUpdate_(cfg, 'config/disparo_contagem', { data: hoje });
    w.update.fields.porVendedor = fsValorRico_(porVendedor);
    w.updateMask.fieldPaths.push('porVendedor');
    fsCommit_(cfg, [w]);
    try { disparoAtualizarPlanilha(); } catch (e) { console.error("Planilha do disparo: " + e.message); }
  } finally {
    lock.releaseLock();
  }
}

// Passo seguinte (2026-10-29): para cada campanha ativa com mais de um passo,
// quem recebeu o passo N há `diasEspera` do passo N+1, sem nenhuma interação nem
// saída em qualquer doc do telefone na campanha, ganha o doc do passo N+1 em
// "fila". Roda antes do envio, na mesma rodada, e não duplica: o doc só nasce
// se ainda não existir.
function disparoAgendarPassos_(cfg, campanhas) {
  var agora = Date.now();
  Object.keys(campanhas).forEach(function (idCamp) {
    var camp = campanhas[idCamp], passos = camp.passos || [];
    if (camp.status !== 'ativa' || passos.length < 2) return;
    var docs = fsConsulta_(cfg, 'disparos', 'campanha', idCamp, 5000);
    var existe = {}, interagiu = {};
    docs.forEach(function (x) {
      existe[x._id] = true;
      // Quem interagiu, saiu, falhou ou foi tirado da campanha não recebe o passo seguinte.
      if (x.interagiuEm || x.tocouEm || x.respondeuEm || x.removidoEm || x.status === 'saiu' || x.status === 'erro') interagiu[x.telFim8] = true;
    });
    docs.forEach(function (d) {
      var n = Number(d.passo || 1);
      if (d.status !== 'enviado' || d.teste || interagiu[d.telFim8] || n >= passos.length) return;
      var prox = passos[n]; // passos[N] é o passo N+1 (a lista começa em 0)
      var modelo = prox && prox.modelos && prox.modelos[d.segmento];
      if (!modelo) return;
      if (!d.enviadoEm || agora < d.enviadoEm + Number(prox.diasEspera || 0) * 86400000) return;
      // Id antigo sem sufixo (r1-abc123) vira r1-abc123-p2; com sufixo troca o número.
      var idProx = (/-p\d+$/.test(d._id) ? d._id.replace(/-p\d+$/, '') : d._id) + '-p' + (n + 1);
      if (existe[idProx]) return;
      var novo = {
        campanha: idCamp, idOportunidade: d.idOportunidade || '', idCliente: d.idCliente || '',
        idVendedor: d.idVendedor, nomeVendedor: d.nomeVendedor || '', telVendedor: d.telVendedor || '',
        nomeLead: d.nomeLead || '', nomeLeadCompleto: d.nomeLeadCompleto || '', telLead: d.telLead, telFim8: d.telFim8,
        segmento: d.segmento || '', referencia: d.referencia || '', modelo: modelo, passo: n + 1,
        status: 'fila', ordem: d.ordem || 0, criadoEm: agora
      };
      if (fsCommit_(cfg, [fsUpdate_(cfg, 'disparos/' + idProx, novo, { exists: false })])) existe[idProx] = true;
    });
  });
}

// Quantos parâmetros do corpo o modelo usa vem de modelos_disparo/{modelo}.variaveis
// (nome, referência, prazo); sem o doc, vão os 3 de sempre.
function disparoMandarModelo_(props, d, prazo, modelo) {
  var sufixo = d._id + '.' + d.telVendedor + '.' + disparoAssinatura_(d._id, d.telVendedor);
  var parametros = [d.nomeLead, d.referencia, prazo || props.prazo].map(function (t) { return { type: 'text', text: t }; });
  var variaveis = Number(modelo && modelo.variaveis);
  if (variaveis >= 1 && variaveis <= 3) parametros = parametros.slice(0, variaveis);
  var corpo = {
    messaging_product: 'whatsapp', to: d.telLead, type: 'template',
    template: {
      name: d.modelo, language: { code: 'pt_BR' },
      components: [
        { type: 'body', parameters: parametros },
        { type: 'button', sub_type: 'url', index: '0', parameters: [{ type: 'text', text: sufixo }] },
        { type: 'button', sub_type: 'quick_reply', index: '1', parameters: [{ type: 'payload', payload: 'SAIR:' + d._id }] }
      ]
    }
  };
  return disparoGraph_(props, corpo);
}

function disparoGraph_(props, corpo) {
  var resp = UrlFetchApp.fetch('https://graph.facebook.com/' + props.versao + '/' + props.phoneId + '/messages', {
    method: 'post', contentType: 'application/json', muteHttpExceptions: true,
    headers: { Authorization: 'Bearer ' + props.token }, payload: JSON.stringify(corpo)
  });
  var txt = resp.getContentText(), json = {};
  try { json = JSON.parse(txt); } catch (e) { /* resposta fora do padrão: vai como texto no erro */ }
  if (resp.getResponseCode() === 200 && json.messages && json.messages[0]) return { ok: true, wamid: json.messages[0].id };
  var err = json.error || {};
  return { ok: false, codigo: Number(err.code || 0), erro: String(err.message || txt).slice(0, 300) };
}

// Mensagem de texto livre: só vale dentro das 24h depois que o lead escreveu.
function disparoMandarTexto_(props, para, texto) {
  return disparoGraph_(props, { messaging_product: 'whatsapp', to: para, type: 'text', text: { body: texto, preview_url: true } });
}

// Manda o texto e guarda a mensagem de saída em mensagens_disparo.
function disparoEnviarTexto_(cfg, props, d, para, texto) {
  var r = disparoMandarTexto_(props, para, texto);
  if (r.ok) disparoGravarMensagem_(cfg, d, 'saida', r.wamid, 'text', texto);
  return r;
}

// Chamada genérica à Graph API (modelos e mídia). O token só vai no cabeçalho.
function disparoMetaPedir_(props, metodo, caminho, corpo) {
  var opcoes = { method: metodo, muteHttpExceptions: true, headers: { Authorization: 'Bearer ' + props.token } };
  if (corpo) { opcoes.contentType = 'application/json'; opcoes.payload = JSON.stringify(corpo); }
  var resp = UrlFetchApp.fetch('https://graph.facebook.com/' + props.versao + '/' + caminho, opcoes);
  var json = {};
  try { json = JSON.parse(resp.getContentText()); } catch (e) { /* resposta fora do padrão */ }
  return { codigo: resp.getResponseCode(), json: json };
}

// Guarda a mensagem em mensagens_disparo/{in|out}-<wamid>. Devolve o id, ou ''
// se ela já estava lá (a Meta repete o webhook).
function disparoGravarMensagem_(cfg, d, direcao, wamid, tipo, texto) {
  var id = (direcao === 'entrada' ? 'in-' : 'out-') + String(wamid || Utilities.getUuid()).replace(/[^A-Za-z0-9_-]/g, '');
  var criou = fsCommit_(cfg, [fsUpdateRico_(cfg, 'mensagens_disparo/' + id, {
    telFim8: d.telFim8 || disparoFim8_(d.telLead), campanha: d.campanha || '', disparo: d._id, idVendedor: String(d.idVendedor || ''),
    nomeLead: d.nomeLeadCompleto || d.nomeLead || '', direcao: direcao, tipo: tipo || 'text',
    texto: String(texto || '').slice(0, 2000), em: Date.now()
  }, { exists: false })]);
  return criou ? id : '';
}

// Pasta do Drive das mídias recebidas: a da Script Property, ou uma nova.
function disparoPastaMidia_() {
  var p = PropertiesService.getScriptProperties();
  var id = p.getProperty('DISPARO_PASTA_MIDIA');
  if (id) return DriveApp.getFolderById(id);
  var pasta = DriveApp.createFolder('Solar Green - Disparo mídias');
  p.setProperty('DISPARO_PASTA_MIDIA', pasta.getId());
  return pasta;
}

// O link da mídia na Meta expira, então ela é baixada na hora e vai pro Drive.
function disparoSalvarMidia_(props, info) {
  var meta = disparoMetaPedir_(props, 'get', info.id);
  if (meta.codigo !== 200 || !meta.json.url) throw new Error('A Meta não devolveu o link da mídia (' + meta.codigo + ').');
  var resp = UrlFetchApp.fetch(meta.json.url, { muteHttpExceptions: true, headers: { Authorization: 'Bearer ' + props.token } });
  if (resp.getResponseCode() !== 200) throw new Error('A Meta recusou o download da mídia (' + resp.getResponseCode() + ').');
  var mime = String(meta.json.mime_type || info.mime_type || '');
  var nome = String(info.filename || (info.id + '.' + (mime.split('/')[1] || 'bin').split(';')[0]));
  var blob = resp.getBlob().setName(nome);
  if (mime) blob.setContentType(mime);
  return { driveId: disparoPastaMidia_().createFile(blob).getId(), mime: mime, nomeArquivo: nome };
}

// ── Toque no botão ─────────────────────────────────────────────────────

function actionDisparoToque(body) {
  if (!disparoSegredoOk_(body)) return { ok: false, erro: 'não autorizado' };
  var token = String(body.token || '');
  if (!/^[A-Za-z0-9_-]{1,80}$/.test(token)) return { ok: false, erro: 'token inválido' };
  var cfg = fbConfig_();
  if (!cfg) return { ok: false, erro: 'FIREBASE_SERVICE_ACCOUNT não configurada' };
  var doc = fsGet_(cfg, 'disparos/' + token);
  if (!doc) return { ok: false, erro: 'disparo não encontrado' };
  var d = doc.dados; d._id = token;
  return disparoRegistrar_(cfg, d, 'toque', '', 'toque-' + token);
}

// ── Mensagens que chegam no número do disparo ──────────────────────────

function actionDisparoWebhook(body) {
  if (!disparoSegredoOk_(body)) return { ok: false, erro: 'não autorizado' };
  var cfg = fbConfig_();
  if (!cfg) return { ok: false, erro: 'FIREBASE_SERVICE_ACCOUNT não configurada' };
  var props = disparoProps_();
  var feitos = 0;
  ((body.payload || {}).entry || []).forEach(function (en) {
    (en.changes || []).forEach(function (ch) {
      var v = ch.value || {};
      if (ch.field === 'message_template_status_update') { disparoTratarStatusModelo_(cfg, v); return; }
      (v.messages || []).forEach(function (m) { if (disparoTratarMensagem_(cfg, props, m)) feitos++; });
      (v.statuses || []).forEach(function (s) { disparoTratarStatus_(cfg, s); });
    });
  });
  return { ok: true, interacoes: feitos };
}

// A Meta aprovou, recusou ou pausou um modelo: o status e o motivo vão pro doc.
function disparoTratarStatusModelo_(cfg, v) {
  var nome = String(v.message_template_name || '');
  if (!/^[A-Za-z0-9_]{1,512}$/.test(nome) || !v.event) return;
  fsCommit_(cfg, [fsUpdate_(cfg, 'modelos_disparo/' + nome, {
    nome: nome, status: String(v.event).toUpperCase(),
    motivo: v.reason && v.reason !== 'NONE' ? String(v.reason).slice(0, 500) : '', atualizadoEm: Date.now()
  })]);
}

function disparoTratarMensagem_(cfg, props, m) {
  var fim8 = disparoFim8_(m.from);
  var docsTel = fsConsulta_(cfg, 'disparos', 'telFim8', fim8, 30);
  var d = null;
  if (m.context && m.context.id) d = fsConsulta_(cfg, 'disparos', 'wamid', m.context.id, 1)[0] || null;
  if (!d) {
    // Sem citação: o disparo mais recente pra esse telefone.
    d = docsTel.slice().sort(function (a, b) {
      return (b.enviadoEm || 0) - (a.enviadoEm || 0);
    })[0] || null;
  }
  if (!d) { console.log('Mensagem no número do disparo sem disparo correspondente (final ' + fim8.slice(-4) + ').'); return false; }
  // Um objeto só por disparo, pra o que for gravado aqui valer pro resto da função.
  var igual = docsTel.filter(function (x) { return x._id === d._id; })[0];
  if (igual) d = igual; else docsTel.push(d);
  var daCampanha = docsTel.filter(function (x) { return (x.campanha || '') === (d.campanha || ''); });

  // Toda mensagem do lead é guardada (a tela de conversa lê daqui), mídia inclusa.
  // Erro na mídia não derruba o registro: a mensagem fica com midiaErro.
  var tipo = m.type || 'text';
  var info = m[tipo] && typeof m[tipo] === 'object' ? m[tipo] : {};
  var texto = tipo === 'text' ? (m.text || {}).body : tipo === 'button' ? info.text : info.caption;
  var idMsg = disparoGravarMensagem_(cfg, d, 'entrada', m.id, tipo, texto);
  if (!idMsg) return false; // a Meta repetiu a entrega: já foi tratada
  if (info.id) {
    var anexo = {};
    try { anexo.midia = disparoSalvarMidia_(props, info); } catch (e) {
      anexo.midiaErro = String(e.message).slice(0, 300);
      console.error('Mídia do disparo: ' + anexo.midiaErro);
    }
    fsCommit_(cfg, [fsUpdateRico_(cfg, 'mensagens_disparo/' + idMsg, anexo)]);
  }

  // Robô de atendimento do outro lado (Chumber Contabilidade, 28/10): ele responde
  // a nossa resposta e os dois ficavam conversando sem parar. Antes isso era cortado
  // ignorando tudo depois da 1ª mensagem do lead, o que também escondia a conversa
  // de gente de verdade. Agora tudo é gravado, a resposta automática sai uma vez só
  // (respondeuEm) e o telefone que manda mais de 5 mensagens em 2 minutos fica
  // marcado como robô (robo), o que corta o aviso sem perder nada do que ele escreveu.
  if (!d.robo && daCampanha.some(function (x) { return x.robo; })) d.robo = true;
  if (!d.robo) disparoVerificarRobo_(cfg, fim8, daCampanha);

  var payload = m.button && m.button.payload ? String(m.button.payload) : '';
  var botao = m.button ? String(m.button.text || '') : '';
  if (payload.indexOf('SAIR') === 0 || /n[ãa]o (quero|tenho interesse)/i.test(botao)) {
    // "Não tenho interesse agora" (modelo novo) sai só da campanha; "Não quero
    // receber" (modelo antigo) e botão de sair sem texto conhecido saem de tudo.
    disparoSair_(cfg, props, m, d, daCampanha, /n[ãa]o tenho interesse/i.test(botao));
    return false;
  }

  var agora = Date.now();
  // A primeira interação sempre avisa; as mensagens seguintes, no máximo 1 por
  // hora por disparo, e nunca se o telefone for robô.
  if (!d.interagiuEm || (!d.robo && agora - Number(d.ultimoAvisoEm || 0) >= DISPARO_AVISO_INTERVALO_MS)) {
    disparoRegistrar_(cfg, d, 'resposta', texto || '[' + tipo + ']', 'msg-' + String(m.id || '').replace(/[^A-Za-z0-9_-]/g, '').slice(-60));
  }
  // Resposta automática: uma vez por contato e campanha.
  if (!daCampanha.some(function (x) { return x.respondeuEm; })) {
    fsCommit_(cfg, [fsUpdate_(cfg, 'disparos/' + d._id, { respondeuEm: agora })]);
    d.respondeuEm = agora;
    var link = 'https://wa.me/' + d.telVendedor + '?text=' + encodeURIComponent(DISPARO_TEXTO_VENDEDOR);
    disparoEnviarTexto_(cfg, props, d, m.from, 'Recebi, ' + d.nomeLead + '. Quem vai te atender é ' + d.nomeVendedor +
      ', da Solar Green. Toque aqui pra falar direto: ' + link);
  }
  return true;
}

// Mais de DISPARO_ROBO_MENSAGENS mensagens do mesmo telefone em 2 minutos: marca
// todos os disparos dele na campanha. A janela é filtrada aqui no código porque
// a consulta do Firestore usa um filtro só.
function disparoVerificarRobo_(cfg, fim8, daCampanha) {
  var agora = Date.now();
  var recentes = fsConsulta_(cfg, 'mensagens_disparo', 'telFim8', fim8, 500).filter(function (x) {
    return x.direcao === 'entrada' && x.em >= agora - DISPARO_ROBO_JANELA_MS;
  });
  if (recentes.length <= DISPARO_ROBO_MENSAGENS) return;
  fsCommit_(cfg, daCampanha.map(function (x) { return fsUpdate_(cfg, 'disparos/' + x._id, { robo: true, roboEm: agora }); }));
  daCampanha.forEach(function (x) { x.robo = true; x.roboEm = agora; });
}

// Botão de sair: todos os disparos do telefone na campanha viram "saiu". Sem
// soCampanha, vale também pras outras campanhas (opt-out global). O lead vai
// pra etapa de papel perdido do Comercial nos dois casos.
function disparoSair_(cfg, props, m, d, daCampanha, soCampanha) {
  var agora = Date.now();
  var writes = daCampanha.map(function (x) { return fsUpdate_(cfg, 'disparos/' + x._id, { status: 'saiu', saiuEm: agora }); });
  if (!soCampanha) writes.push(fsUpdate_(cfg, 'disparo_optout/' + disparoFim8_(m.from), { em: agora, disparo: d._id }));
  fsCommit_(cfg, writes);
  if (!d.teste) {
    try { disparoMoverParaPerdido_(cfg, d); } catch (e) { console.error('Lead do disparo que saiu: ' + e.message); }
  }
  disparoEnviarTexto_(cfg, props, d, m.from, DISPARO_TEXTO_SAIDA);
}

function disparoTratarStatus_(cfg, s) {
  if (s.status !== 'failed' && s.status !== 'delivered' && s.status !== 'read') return;
  var d = fsConsulta_(cfg, 'disparos', 'wamid', s.id, 1)[0];
  if (!d) return;
  var campos = {};
  if (s.status === 'delivered') campos.entregueEm = Date.now();
  if (s.status === 'read') campos.lidoEm = Date.now();
  if (s.status === 'failed') {
    var e = (s.errors || [])[0] || {};
    if (Number(e.code) === DISPARO_ERRO_PAGAMENTO) {
      // A Meta aceitou e depois recusou por pagamento: volta pra fila (enviadoEm
      // zerado pra não contar no limite de 24h) e o envio espera 1 hora.
      disparoPagamentoFalhou_(cfg);
      campos = { status: 'fila', enviadoEm: 0, wamid: '', erro: 'Pagamento recusado pela Meta (131042); voltou pra fila.' };
    } else {
      campos.status = 'erro'; campos.erroCodigo = Number(e.code || 0); campos.erro = String(e.title || e.message || 'falhou').slice(0, 300);
    }
  }
  fsCommit_(cfg, [fsUpdate_(cfg, 'disparos/' + d._id, campos)]);
}

var DISPARO_ERRO_PAGAMENTO = 131042;

function disparoPagamentoFalhou_(cfg) {
  fsCommit_(cfg, [fsUpdate_(cfg, 'config/disparo_pagamento', { falhouEm: Date.now() })]);
}

function disparoFilaDaCampanha_(cfg, idCampanha, limite) {
  var resp = fsPedir_(cfg, 'post', cfg.base + ':runQuery', { structuredQuery: {
    from: [{ collectionId: 'disparos' }],
    where: { compositeFilter: { op: 'AND', filters: [
      { fieldFilter: { field: { fieldPath: 'status' }, op: 'EQUAL', value: fsValor_('fila') } },
      { fieldFilter: { field: { fieldPath: 'campanha' }, op: 'EQUAL', value: fsValor_(idCampanha) } }
    ] } },
    limit: limite
  } });
  if (resp.getResponseCode() !== 200) throw new Error('Firestore recusou a fila da campanha ' + idCampanha + ': ' + resp.getContentText().slice(0, 200));
  return JSON.parse(resp.getContentText()).filter(function (r) { return r.document; }).map(function (r) {
    var item = fsLerCampos_(r.document.fields || {});
    item._id = r.document.name.split('/').pop();
    return item;
  });
}

// ── Interação: antessala, funil e push ─────────────────────────────────

function disparoRegistrar_(cfg, d, tipo, texto, idInteracao) {
  var agora = Date.now();
  var vendedores = vendedoresFs_(cfg), nomes = {}, gestao = [];
  vendedores.forEach(function (v) {
    nomes[String(v.IdVendedor)] = v.Nome || '';
    if (ehGestaoAtivaFs_(v)) gestao.push(String(v.IdVendedor));
  });
  var registro = {
    id: idInteracao, disparo: d._id, tipo: tipo, texto: String(texto || '').slice(0, 500),
    idOportunidade: d.idOportunidade || '', idVendedor: String(d.idVendedor), nomeLead: d.nomeLeadCompleto || d.nomeLead,
    segmento: d.segmento || '', criadoEmMs: agora
  };
  // Mesmo id duas vezes (o Worker ou a Meta repetiram) não cria nem avisa de novo.
  var criou = fsCommit_(cfg, [fsUpdate_(cfg, 'interacoes_disparo/' + idInteracao, registro, { exists: false })]);
  if (!criou) return { ok: true, acao: 'já registrado' };

  var campos = { ultimoAvisoEm: agora };
  var writes = [];
  if (!d.interagiuEm) {
    campos.interagiuEm = agora;
    if (!d.teste) {
      if (d.idOportunidade) {
        disparoMexerNoFunil_(cfg, d, tipo, texto, nomes);
      } else {
        // Contato sem funil: a primeira resposta ou toque abre o lead em Retomar Contato.
        var pipe = disparoPipelineComercial_(cfg);
        var idLead = pipe ? disparoCriarLead_(cfg, d, pipe, DISPARO_ETAPA_RETORNO, '') : '';
        if (idLead) {
          campos.idOportunidade = idLead;
          writes.push(fsUpdate_(cfg, 'interacoes_disparo/' + idInteracao, { idOportunidade: idLead }));
        }
      }
    }
  }
  if (tipo === 'toque' && !d.tocouEm) campos.tocouEm = agora;
  writes.push(fsUpdate_(cfg, 'disparos/' + d._id, campos));
  fsCommit_(cfg, writes);
  Object.keys(campos).forEach(function (k) { d[k] = campos[k]; });

  var titulo = registro.nomeLead + ' interagiu com o disparo';
  var corpo = (tipo === 'toque' ? 'Tocou no botão e foi pro seu WhatsApp' : 'Respondeu: "' + registro.texto.slice(0, 80) + '"') +
    (d.segmento ? ' · ' + d.segmento : '');
  var url = 'index.html?lead=' + (d.idOportunidade || '') + '&notif=disparo:' + idInteracao;
  var vendedor = String(d.idVendedor);
  enviarPush_(cfg, [vendedor], { titulo: titulo, corpo: corpo, url: url, tag: 'disparo-' + d._id, notifId: 'disparo:' + idInteracao });
  enviarPush_(cfg, gestao.filter(function (g) { return g !== vendedor; }), {
    titulo: titulo, corpo: corpo + ' · para ' + (nomes[vendedor] || 'vendedor'), url: url, tag: 'disparo-' + d._id, notifId: 'disparo:' + idInteracao
  });
  return { ok: true, acao: 'registrado' };
}

function disparoMexerNoFunil_(cfg, d, tipo, texto, nomes) {
  var caminho = 'funil/' + d.idOportunidade;
  var doc = fsGet_(cfg, caminho);
  if (!doc) return;
  var f = doc.dados, agoraIso = new Date().toISOString();
  var campos = {}, extras = [];
  if (String(f.IdVendedor || '') !== String(d.idVendedor)) {
    campos.IdVendedor = String(d.idVendedor);
    campos.VendedorAnterior = nomes[String(f.IdVendedor || '')] || String(f.IdVendedor || '');
  }
  if (f.Etapa !== DISPARO_ETAPA_RETORNO && DISPARO_ETAPAS_QUE_FICAM.indexOf(f.Etapa) === -1) {
    campos.Etapa = DISPARO_ETAPA_RETORNO;
    extras.push({ fieldPath: 'EtapasPassadas', appendMissingElements: { values: [fsValorRico_(DISPARO_ETAPA_RETORNO)] } });
    extras.push({ fieldPath: 'Transicoes', appendMissingElements: { values: [fsValorRico_({ Etapa: DISPARO_ETAPA_RETORNO, Em: agoraIso })] } });
  }
  var obs = tipo === 'toque' ? 'Disparo de reativação: tocou no botão e foi pro WhatsApp do vendedor.'
    : 'Disparo de reativação: respondeu "' + String(texto || '').slice(0, 200) + '".';
  extras.push({ fieldPath: 'Atividades', appendMissingElements: { values: [fsValorRico_({ Tipo: 'Conversa no WhatsApp', Em: agoraIso, Observacao: obs })] } });
  var w = fsUpdate_(cfg, caminho, campos, { exists: true });
  w.updateTransforms = extras;
  fsCommit_(cfg, [w]);
}

// Pipeline Comercial e o papel de cada etapa, do mesmo jeito que o
// _tools/preparar-disparo.js: pelo nome do pipeline e pelo campo Papel da etapa.
function disparoPipelineComercial_(cfg) {
  var achado = fsListar_(cfg, 'funil_pipelines').filter(function (p) {
    return String(p.Nome || '').toLowerCase() === 'comercial';
  })[0];
  if (!achado) { console.error('Pipeline Comercial não encontrado.'); return null; }
  var pipe = { id: achado._id, papel: {}, perdido: '' };
  (achado.Etapas || []).forEach(function (e) {
    pipe.papel[e.Nome] = e.Papel || '';
    if (e.Papel === 'perdido' && !pipe.perdido) pipe.perdido = e.Nome;
  });
  return pipe;
}

// Lead novo no Comercial, com os campos que o preparar-disparo.js grava para
// cliente sem funil. Devolve o id, ou '' se não conseguiu criar.
function disparoCriarLead_(cfg, d, pipe, etapa, motivoPerda) {
  var agora = new Date();
  var id = Utilities.getUuid().replace(/-/g, '').substring(0, 8);
  var w = fsUpdateRico_(cfg, 'funil/' + id, {
    IdOportunidade: id, IdCliente: d.idCliente || '', IdVendedor: String(d.idVendedor), IdServico: DISPARO_SERVICO_NOVO,
    Etapa: etapa, Observacoes: 'Lead criado pelo disparo de reativação (' + Utilities.formatDate(agora, 'America/Belem', 'dd/MM/yyyy') + ').',
    'Valor Estimado': DISPARO_VALOR_NOVO, MotivoPerda: motivoPerda || '', NomeCliente: d.nomeLeadCompleto || d.nomeLead || '',
    TelefoneCliente: d.telLead, Pipeline: pipe.id, DataCriacao: Utilities.formatDate(agora, 'America/Belem', 'yyyy-MM-dd'),
    EtapasPassadas: [etapa], Transicoes: [{ Etapa: etapa, Em: agora.toISOString() }]
  }, { exists: false });
  w.update.fields.CriadoEm = { timestampValue: agora.toISOString() };
  w.updateMask.fieldPaths.push('CriadoEm');
  return fsCommit_(cfg, [w]) ? id : '';
}

// Quem saiu vai pra etapa de papel perdido do Comercial: move o lead que já
// existe ou cria um lá. Venda fechada (papel ganho) não é mexida.
function disparoMoverParaPerdido_(cfg, d) {
  var pipe = disparoPipelineComercial_(cfg);
  if (!pipe || !pipe.perdido) { console.error('Comercial sem etapa de papel perdido: lead do disparo não movido.'); return; }
  if (!d.idOportunidade) {
    var idLead = disparoCriarLead_(cfg, d, pipe, pipe.perdido, DISPARO_MOTIVO_PERDA);
    if (idLead) { fsCommit_(cfg, [fsUpdate_(cfg, 'disparos/' + d._id, { idOportunidade: idLead })]); d.idOportunidade = idLead; }
    return;
  }
  var doc = fsGet_(cfg, 'funil/' + d.idOportunidade);
  if (!doc || doc.dados.Etapa === pipe.perdido || pipe.papel[doc.dados.Etapa] === 'ganho') return;
  var agoraIso = new Date().toISOString();
  var w = fsUpdate_(cfg, 'funil/' + d.idOportunidade, { Etapa: pipe.perdido, MotivoPerda: DISPARO_MOTIVO_PERDA }, { exists: true });
  w.updateTransforms = [
    { fieldPath: 'EtapasPassadas', appendMissingElements: { values: [fsValorRico_(pipe.perdido)] } },
    { fieldPath: 'Transicoes', appendMissingElements: { values: [fsValorRico_({ Etapa: pipe.perdido, Em: agoraIso })] } },
    { fieldPath: 'Atividades', appendMissingElements: { values: [fsValorRico_({ Tipo: 'Conversa no WhatsApp', Em: agoraIso, Observacao: 'Disparo de reativação: tocou em "Não tenho interesse agora".' })] } }
  ];
  fsCommit_(cfg, [w]);
}

var DISPARO_BOTAO_URL = 'Falar com o atendente';
var DISPARO_BOTAO_SAIR = 'Não tenho interesse agora';

// Confere o idToken do Firebase em accounts:lookup e acha a pessoa em
// vendedores (o id do documento é o uid; senão busca por IdVendedor). Só
// usuário ativo passa; com exigirGestao, só os papéis de gestão.
function disparoAutenticar_(cfg, body, exigirGestao) {
  var idToken = String(body.idToken || '');
  if (!idToken) return { erro: 'não autorizado' };
  var resp = UrlFetchApp.fetch('https://identitytoolkit.googleapis.com/v1/accounts:lookup?key=' + DISPARO_FIREBASE_API_KEY, {
    method: 'post', contentType: 'application/json', muteHttpExceptions: true, payload: JSON.stringify({ idToken: idToken })
  });
  if (resp.getResponseCode() !== 200) return { erro: 'não autorizado' };
  var uid = '';
  try { uid = String(JSON.parse(resp.getContentText()).users[0].localId || ''); } catch (e) { /* sem usuário na resposta */ }
  if (!uid) return { erro: 'não autorizado' };
  var doc = fsGet_(cfg, 'vendedores/' + uid);
  var v = doc ? doc.dados : fsConsulta_(cfg, 'vendedores', 'IdVendedor', uid, 1)[0];
  if (!v || String(v.Status || '').trim() !== 'Ativo') return { erro: 'não autorizado' };
  if (exigirGestao && !ehGestaoAtivaFs_(v)) return { erro: 'sem permissão' };
  return { vendedor: v, uid: uid };
}

// Todos os modelos da conta na Meta (a resposta vem paginada).
function disparoListarModelosMeta_(props, campos) {
  var lista = [], depois = '';
  for (var pagina = 0; pagina < 10; pagina++) {
    var r = disparoMetaPedir_(props, 'get', props.wabaId + '/message_templates?fields=' + campos + '&limit=100' +
      (depois ? '&after=' + encodeURIComponent(depois) : ''));
    if (r.codigo !== 200) throw new Error('A Meta recusou a lista de modelos (' + r.codigo + '): ' + String((r.json.error || {}).message || '').slice(0, 200));
    lista = lista.concat(r.json.data || []);
    var pag = r.json.paging || {};
    depois = (pag.cursors || {}).after || '';
    if (!pag.next || !depois) break;
  }
  return lista;
}

// Endereço do botão URL dos modelos que já existem (o do Worker, com {{1}} no fim).
function disparoEnderecoBotao_(props) {
  var achado = '';
  disparoListarModelosMeta_(props, 'name,components').forEach(function (t) {
    (t.components || []).forEach(function (c) {
      (c.buttons || []).forEach(function (b) { if (!achado && b.type === 'URL' && /\{\{1\}\}/.test(b.url || '')) achado = b.url; });
    });
  });
  return achado;
}

function actionDisparoCriarModelo(body) {
  var cfg = fbConfig_();
  if (!cfg) return { ok: false, erro: 'FIREBASE_SERVICE_ACCOUNT não configurada' };
  var quem = disparoAutenticar_(cfg, body, true);
  if (quem.erro) return { ok: false, erro: quem.erro };
  var props = disparoProps_();
  if (!props.token || !props.wabaId) return { ok: false, erro: 'Faltam WA_TOKEN ou WA_WABA_ID nas Script Properties.' };

  var nome = String(body.nome || '').trim(), corpo = String(body.corpo || '').trim();
  if (!/^[a-z0-9_]{3,60}$/.test(nome)) return { ok: false, erro: 'Nome inválido: use de 3 a 60 letras minúsculas, números ou _.' };
  if (!corpo || corpo.length > 1024) return { ok: false, erro: 'O texto precisa ter de 1 a 1024 caracteres.' };
  var ordem = [];
  (corpo.match(/\{\{\d+\}\}/g) || []).forEach(function (x) {
    var n = Number(x.replace(/\D/g, ''));
    if (ordem.indexOf(n) === -1) ordem.push(n);
  });
  if (!ordem.length || ordem.length > 3 || ordem.some(function (n, i) { return n !== i + 1; })) {
    return { ok: false, erro: 'O texto precisa ter {{1}} e no máximo {{3}}, em ordem.' };
  }

  var base;
  try { base = disparoEnderecoBotao_(props); } catch (e) { return { ok: false, erro: e.message }; }
  if (!base) return { ok: false, erro: 'Não achei o endereço do botão "' + DISPARO_BOTAO_URL + '" nos modelos atuais da Meta. Nada foi criado.' };

  var r = disparoMetaPedir_(props, 'post', props.wabaId + '/message_templates', {
    name: nome, category: 'MARKETING', language: 'pt_BR',
    components: [
      { type: 'BODY', text: corpo, example: { body_text: [['Maria', 'outubro', 'sexta-feira, 31/10'].slice(0, ordem.length)] } },
      { type: 'BUTTONS', buttons: [
        { type: 'URL', text: DISPARO_BOTAO_URL, url: base, example: [base.replace('{{1}}', 'exemplo')] },
        { type: 'QUICK_REPLY', text: DISPARO_BOTAO_SAIR }
      ] }
    ]
  });
  if (r.codigo !== 200 || !r.json.id) {
    var err = r.json.error || {};
    return { ok: false, erro: 'A Meta recusou o modelo: ' + String(err.error_user_msg || err.message || r.codigo).slice(0, 300) };
  }
  var agora = Date.now(), status = String(r.json.status || 'PENDING');
  fsCommit_(cfg, [fsUpdateRico_(cfg, 'modelos_disparo/' + nome, {
    nome: nome, idioma: 'pt_BR', categoria: 'MARKETING', corpo: corpo, variaveis: ordem.length,
    botoes: [{ tipo: 'URL', texto: DISPARO_BOTAO_URL }, { tipo: 'QUICK_REPLY', texto: DISPARO_BOTAO_SAIR }],
    status: status, motivo: '', metaId: String(r.json.id), criadoEm: agora, atualizadoEm: agora
  })]);
  return { ok: true, nome: nome, status: status };
}

// Copia pra modelos_disparo todos os modelos da conta na Meta.
function disparoSincronizarModelosAgora_(cfg, props) {
  var existentes = {};
  fsListar_(cfg, 'modelos_disparo').forEach(function (x) { existentes[x._id] = true; });
  var agora = Date.now();
  var writes = disparoListarModelosMeta_(props, 'name,status,language,category,components,rejected_reason').filter(function (t) {
    return /^[A-Za-z0-9_]+$/.test(t.name || '');
  }).map(function (t) {
    var corpo = '', variaveis = 0, botoes = [];
    (t.components || []).forEach(function (c) {
      if (c.type === 'BODY') {
        corpo = c.text || '';
        (corpo.match(/\{\{\d+\}\}/g) || []).forEach(function (x) { variaveis = Math.max(variaveis, Number(x.replace(/\D/g, ''))); });
      }
      if (c.type === 'BUTTONS') botoes = (c.buttons || []).map(function (b) { return { tipo: b.type, texto: b.text || '' }; });
    });
    var campos = {
      nome: t.name, idioma: t.language || '', categoria: t.category || '', corpo: corpo, variaveis: variaveis, botoes: botoes,
      status: t.status || '', motivo: t.rejected_reason && t.rejected_reason !== 'NONE' ? String(t.rejected_reason) : '',
      metaId: String(t.id || ''), atualizadoEm: agora
    };
    if (!existentes[t.name]) campos.criadoEm = agora;
    return fsUpdateRico_(cfg, 'modelos_disparo/' + t.name, campos);
  });
  for (var i = 0; i < writes.length; i += 100) fsCommit_(cfg, writes.slice(i, i + 100));
  return writes.length;
}

function actionDisparoSincronizarModelos(body) {
  var cfg = fbConfig_();
  if (!cfg) return { ok: false, erro: 'FIREBASE_SERVICE_ACCOUNT não configurada' };
  var quem = disparoAutenticar_(cfg, body, true);
  if (quem.erro) return { ok: false, erro: quem.erro };
  var props = disparoProps_();
  if (!props.token || !props.wabaId) return { ok: false, erro: 'Faltam WA_TOKEN ou WA_WABA_ID nas Script Properties.' };
  try { return { ok: true, modelos: disparoSincronizarModelosAgora_(cfg, props) }; } catch (e) { return { ok: false, erro: e.message }; }
}

// Gatilho de 4 em 4 horas (disparoInstalarGatilhoModelos): pega o que a Meta
// mudou sem o webhook avisar. Sem parâmetros, porque o gatilho não passa nenhum.
function disparoSincronizarModelosGatilho() {
  var cfg = fbConfig_(), props = disparoProps_();

// ── Firestore: consulta por um campo e valores com mapa ────────────────

function fsConsulta_(cfg, colecao, campo, valor, limite) {
  var resp = fsPedir_(cfg, 'post', cfg.base + ':runQuery', { structuredQuery: {
    from: [{ collectionId: colecao }],
    where: { fieldFilter: { field: { fieldPath: campo }, op: 'EQUAL', value: fsValor_(valor) } },
    limit: limite || 50
  } });
  if (resp.getResponseCode() !== 200) throw new Error('Firestore recusou a consulta em ' + colecao + ': ' + resp.getContentText().slice(0, 200));
  return JSON.parse(resp.getContentText()).filter(function (r) { return r.document; }).map(function (r) {
    var item = fsLerCampos_(r.document.fields || {});
    item._id = r.document.name.split('/').pop();
    return item;
  });
}

function fsValorRico_(v) {
  if (v && typeof v === 'object' && !Array.isArray(v)) {
    var fields = {};
    Object.keys(v).forEach(function (k) { fields[k] = fsValorRico_(v[k]); });
    return { mapValue: { fields: fields } };
  }
  if (Array.isArray(v)) return { arrayValue: { values: v.map(fsValorRico_) } };
  return fsValor_(v);
}

// Como fsUpdate_, mas aceita mapas e listas nos valores.
function fsUpdateRico_(cfg, caminho, campos, condicao) {
  var w = fsUpdate_(cfg, caminho, {}, condicao);
  Object.keys(campos).forEach(function (k) {
    w.update.fields[k] = fsValorRico_(campos[k]);
    w.updateMask.fieldPaths.push(k);
  });
  return w;
}

// Quantos contatos a Meta conta agora no limite de 24h: enviados nas últimas 24h
// que não falharam (mensagem não entregue não conta pra Meta).
function disparoEnviados24h_(cfg) {
  var resp = fsPedir_(cfg, 'post', cfg.base + ':runQuery', { structuredQuery: {
    from: [{ collectionId: 'disparos' }],
    where: { fieldFilter: { field: { fieldPath: 'enviadoEm' }, op: 'GREATER_THAN_OR_EQUAL', value: fsValor_(Date.now() - 24 * 3600 * 1000) } },
    select: { fields: [{ fieldPath: 'status' }] }
  } });
  if (resp.getResponseCode() !== 200) throw new Error('Firestore recusou a contagem de 24h: ' + resp.getContentText().slice(0, 200));
  return JSON.parse(resp.getContentText()).filter(function (r) {
    return r.document && !(r.document.fields && r.document.fields.status && r.document.fields.status.stringValue === 'erro');
  }).length;
}

// Relatório do disparo numa planilha Google, pedido do Felipe em 28/10: todos
// os leads da campanha com nome, mês/ano do último movimento do funil antes do
// disparo, etapa atual, resposta ao disparo e data de envio. A planilha nasce
// na primeira execução (o id fica em DISPARO_PLANILHA_ID) e é refeita no fim
// de cada rodada do disparoEnviarLote. Custo: ~1.100 leituras por atualização.
function disparoAtualizarPlanilha() {
  var cfg = fbConfig_();
  if (!cfg) return;
  var disparos = fsListar_(cfg, 'disparos').filter(function (d) { return !d.teste; });
  var nomes = {};
  vendedoresFs_(cfg).forEach(function (v) { nomes[String(v.IdVendedor)] = v.Nome || ''; });

  // Quem respondeu conta como "Respondeu" mesmo que antes tenha só clicado.
  var resposta = {};
  fsListar_(cfg, 'interacoes_disparo').forEach(function (x) {
    if (x.tipo === 'resposta' || !resposta[x.disparo]) resposta[x.disparo] = x.tipo;
  });
  function situacao(d) {
    if (d.status === 'saiu') return 'Não quer receber';
    if (d.status === 'fila') return 'Ainda não enviado';
    if (d.status === 'erro') return 'Não entregue';
    if (resposta[d._id] === 'resposta') return 'Respondeu';
    if (resposta[d._id] === 'toque') return 'Clicou';
    return 'Não interagiu';
  }

  // Mesmo critério do preparar-disparo.js: o Em mais recente de Transicoes e
  // Atividades, só que antes da entrada na fila (a interação mexe no funil).
  var funil = fsBatchGet_(cfg, disparos.map(function (d) { return 'funil/' + d.idOportunidade; }),
    ['Etapa', 'IdVendedor', 'Transicoes', 'Atividades']);
  function ultimoAntes(f, limite) {
    return (f.Transicoes || []).concat(f.Atividades || []).reduce(function (m, x) {
      var t = Date.parse(x && x.Em);
      return !isNaN(t) && t < limite ? Math.max(m, t) : m;
    }, 0);
  }

  disparos.sort(function (a, b) {
    return (b.enviadoEm || 0) - (a.enviadoEm || 0) || (a.ordem || 0) - (b.ordem || 0);
  });
  var linhas = disparos.map(function (d) {
    var f = funil['funil/' + d.idOportunidade] || {};
    var ult = d.segmento === 'sem funil' ? 0 : ultimoAntes(f, d.criadoEm || Date.now());
    return [
      d.nomeLeadCompleto || d.nomeLead || '',
      nomes[String(d.idVendedor)] || d.nomeVendedor || '',
      ult ? new Date(ult) : '',
      f.Etapa || '(lead não encontrado)',
      situacao(d),
      d.enviadoEm ? new Date(d.enviadoEm) : ''
    ];
  });

  var props = PropertiesService.getScriptProperties();
  var id = props.getProperty('DISPARO_PLANILHA_ID');
  var ss = id ? SpreadsheetApp.openById(id) : SpreadsheetApp.create('Solar Green - Disparo de reativação');
  if (!id) {
    props.setProperty('DISPARO_PLANILHA_ID', ss.getId());
    ss.setSpreadsheetTimeZone('America/Belem');
  }

  var aba = ss.getSheetByName('Leads') || ss.getSheets()[0].setName('Leads');
  if (aba.getFilter()) aba.getFilter().remove();
  aba.clear();
  var cab = ['Nome', 'Vendedor', 'Último movimento no funil', 'Etapa atual no funil', 'Resposta ao disparo', 'Data de envio'];
  aba.getRange(1, 1, 1, cab.length).setValues([cab]).setFontWeight('bold').setBackground('#003122').setFontColor('#ffffff');
  if (linhas.length) {
    aba.getRange(2, 1, linhas.length, cab.length).setValues(linhas);
    aba.getRange(2, 3, linhas.length, 1).setNumberFormat('mm/yyyy');
    aba.getRange(2, 6, linhas.length, 1).setNumberFormat('dd/mm/yyyy hh:mm');
  }
  aba.setFrozenRows(1);
  aba.getRange(1, 1, linhas.length + 1, cab.length).createFilter();
  aba.autoResizeColumns(1, cab.length);

  var conta = {};
  linhas.forEach(function (l) { conta[l[4]] = (conta[l[4]] || 0) + 1; });
  var resumo = [['Atualizado em', new Date()], ['', ''], ['Resposta ao disparo', 'Leads']];
  ['Clicou', 'Respondeu', 'Não quer receber', 'Não interagiu', 'Não entregue', 'Ainda não enviado'].forEach(function (t) {
    resumo.push([t, conta[t] || 0]);
  });
  resumo.push(['Total', linhas.length]);
  var res = ss.getSheetByName('Resumo') || ss.insertSheet('Resumo');
  res.clear();
  res.getRange(1, 1, resumo.length, 2).setValues(resumo);
  res.getRange(1, 2).setNumberFormat('dd/mm/yyyy hh:mm');
  res.getRange(3, 1, 1, 2).setFontWeight('bold');
  res.getRange(resumo.length, 1, 1, 2).setFontWeight('bold');
  res.autoResizeColumns(1, 2);
  Logger.log('Planilha do disparo atualizada com ' + linhas.length + ' leads: ' + ss.getUrl());
  return ss.getUrl();
}

// Lê vários documentos de uma vez (lotes de 100), só com os campos pedidos.
function fsBatchGet_(cfg, caminhos, campos) {
  var raiz = cfg.base.replace('https://firestore.googleapis.com/v1/', '');
  var out = {};
  for (var i = 0; i < caminhos.length; i += 100) {
    var corpo = { documents: caminhos.slice(i, i + 100).map(function (c) { return raiz + '/' + c; }) };
    if (campos) corpo.mask = { fieldPaths: campos };
    var resp = fsPedir_(cfg, 'post', cfg.base + ':batchGet', corpo);
    if (resp.getResponseCode() !== 200) throw new Error('Firestore recusou a leitura em lote: ' + resp.getContentText().slice(0, 200));
    JSON.parse(resp.getContentText()).forEach(function (r) {
      if (r.found) out[r.found.name.split('/documents/')[1]] = fsLerCampos_(r.found.fields || {});
    });
  }
  return out;
}
